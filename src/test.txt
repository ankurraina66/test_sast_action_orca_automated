
import { describe, expect, it, vi } from "vitest";

import { AppScanService } from "../../src/services/appscan-service.js";
import type {
  AppScanMcpConnector,
  McpTool,
  McpToolCallResult,
} from "../../src/mcp/connector.js";
import type { AppScanProvider } from "../../src/config/config.js";

describe("AppScanService", () => {
  function createMockConnector(
    toolNames: string[] = ["get-apps"],
  ): AppScanMcpConnector {
    return {
      connect: vi.fn().mockResolvedValue(undefined),
      listTools: vi.fn().mockResolvedValue(
        toolNames.map((name) => ({
          name,
          description: `Mock tool ${name}`,
          inputSchema: {
            type: "object",
            properties: {},
          },
        } satisfies McpTool)),
      ),
      callTool: vi.fn().mockResolvedValue({
        content: [{ type: "text", text: "Success" }],
        isError: false,
      } satisfies McpToolCallResult),
      disconnect: vi.fn().mockResolvedValue(undefined),
    };
  }

  function createService(
    connector = createMockConnector(),
    provider: AppScanProvider = "remote",
    connectorFactory = vi.fn<AppScanProvider extends never ? never : (provider: AppScanProvider) => AppScanMcpConnector>(
      () => createMockConnector(),
    ),
  ) {
    return new AppScanService(connector, provider, connectorFactory);
  }

  it("connects through the configured connector", async () => {
    const connector = createMockConnector();
    const service = createService(connector);

    await service.connect();

    expect(connector.connect).toHaveBeenCalledTimes(1);
  });

  it("lists tools through the configured connector", async () => {
    const connector = createMockConnector();
    const service = createService(connector);

    await expect(service.listTools()).resolves.toMatchObject([
      { name: "get-apps" },
    ]);

    expect(connector.listTools).toHaveBeenCalledTimes(1);
  });

  it("delegates tool calls and arguments to the connector", async () => {
    const connector = createMockConnector();
    const service = createService(connector);

    await expect(
      service.callTool("get-apps", { top: 5 }),
    ).resolves.toMatchObject({
      content: [{ type: "text", text: "Success" }],
      isError: false,
    });

    expect(connector.callTool).toHaveBeenCalledWith("get-apps", {
      top: 5,
    });
  });

  it("recognizes the remote get_apps tool name", async () => {
    const connector = createMockConnector(["get_apps"]);
    const service = createService(connector);

    await expect(
      service.findApplicationDiscoveryTool(),
    ).resolves.toMatchObject({ name: "get_apps" });
  });

  it("discovers applications using the available tool name", async () => {
    const connector = createMockConnector();
    const service = createService(connector);

    await service.discoverApplications({ top: 5 });

    expect(connector.callTool).toHaveBeenCalledWith("get-apps", {
      top: 5,
    });
  });

  it("reports when no recognized application discovery tool exists", async () => {
    const connector = createMockConnector(["unrelated-tool"]);
    const service = createService(connector);

    await expect(service.discoverApplications()).rejects.toThrow(
      "does not expose a recognized application discovery tool",
    );
  });

  it("disconnects through the configured connector", async () => {
    const connector = createMockConnector();
    const service = createService(connector);

    await service.disconnect();

    expect(connector.disconnect).toHaveBeenCalledTimes(1);
  });

  it("switches providers only after the new connector is validated", async () => {
    const original = createMockConnector();
    const replacement = createMockConnector([
      "get_apps",
      "get_scans",
    ]);
    const factory = vi.fn(() => replacement);
    const service = createService(original, "remote", factory);

    await expect(service.switchProvider("local")).resolves.toEqual({
      provider: "local",
      toolCount: 2,
    });

    expect(factory).toHaveBeenCalledWith("local");
    expect(replacement.connect).toHaveBeenCalledTimes(1);
    expect(original.disconnect).toHaveBeenCalledTimes(1);
    expect(service.getProvider()).toBe("local");
  });

  it("preserves the current provider if the replacement fails to connect", async () => {
    const original = createMockConnector();
    const replacement = createMockConnector();
    vi.mocked(replacement.connect).mockRejectedValue(
      new Error("Connection refused"),
    );

    const service = createService(
      original,
      "remote",
      () => replacement,
    );

    await expect(service.switchProvider("local")).rejects.toThrow(
      "Could not switch to the local AppScan provider",
    );

    expect(service.getProvider()).toBe("remote");
    expect(original.disconnect).not.toHaveBeenCalled();
    expect(replacement.disconnect).toHaveBeenCalledTimes(1);
  });

  it("preserves the current provider if the replacement exposes no tools", async () => {
    const original = createMockConnector();
    const replacement = createMockConnector([]);
    const service = createService(
      original,
      "remote",
      () => replacement,
    );

    await expect(service.switchProvider("local")).rejects.toThrow(
      "returned no tools",
    );

    expect(service.getProvider()).toBe("remote");
    expect(original.disconnect).not.toHaveBeenCalled();
    expect(replacement.disconnect).toHaveBeenCalledTimes(1);
  });

  it("does not switch when the requested provider is already active", async () => {
    const original = createMockConnector();
    const factory = vi.fn(() => createMockConnector());
    const service = createService(original, "remote", factory);

    await expect(service.switchProvider("remote")).resolves.toEqual({
      provider: "remote",
      toolCount: 1,
    });

    expect(factory).not.toHaveBeenCalled();
    expect(original.disconnect).not.toHaveBeenCalled();
  });

  it("maps tool names case-insensitively", async () => {
    const connector = createMockConnector(["get_apps"]);
    const service = createService(connector);

    await service.callFirstAvailableTool(["GET_APPS"], { top: 3 });

    expect(connector.callTool).toHaveBeenCalledWith("get_apps", {
      top: 3,
    });
  });

  it("rejects calls when the expected provider is not active", async () => {
    const connector = createMockConnector(["create-scan"]);
    const service = createService(connector, "remote");

    await expect(
      service.callFirstAvailableTool(
        ["create-scan"],
        { approved: true },
        "local",
      ),
    ).rejects.toThrow("Provider mismatch");

    expect(connector.callTool).not.toHaveBeenCalled();
  });

  it("does not execute an operation when its tool is unavailable", async () => {
    const connector = createMockConnector(["get-apps"]);
    const service = createService(connector);

    await expect(
      service.callFirstAvailableTool(["create-scan"]),
    ).rejects.toThrow("does not support this operation");

    expect(connector.callTool).not.toHaveBeenCalled();
  });

  it("waits for an in-flight tool call before switching providers", async () => {
    const original = createMockConnector();
    const replacement = createMockConnector(["get_apps"]);

    let finishCall!: (value: McpToolCallResult) => void;
    vi.mocked(original.callTool).mockImplementation(
      () =>
        new Promise<McpToolCallResult>((resolve) => {
          finishCall = resolve;
        }),
    );

    const service = createService(
      original,
      "remote",
      () => replacement,
    );

    const callPromise = service.callTool("get-apps");
    const switchPromise = service.switchProvider("local");

    // Allow queued work to progress; the switch must wait for the call.
    await Promise.resolve();
    expect(replacement.connect).not.toHaveBeenCalled();

    finishCall({
      content: [{ type: "text", text: "Finished" }],
      isError: false,
    });

    await callPromise;
    await expect(switchPromise).resolves.toMatchObject({
      provider: "local",
      toolCount: 1,
    });

    expect(original.callTool).toHaveBeenCalledTimes(1);
    expect(original.disconnect).toHaveBeenCalledTimes(1);
    expect(service.getProvider()).toBe("local");
  });
});
