/**
 * Create a basic configuration for .NET projects
 */
export function createDotNetConfig(targetPath, options) {
    const customBuildInfo = {};
    if (options?.references) {
        customBuildInfo.references = options.references;
    }
    if (options?.configuration) {
        customBuildInfo.configuration = options.configuration;
    }
    return {
        targets: [{
                path: targetPath,
                customBuildInfo: Object.keys(customBuildInfo).length > 0 ? customBuildInfo : undefined,
                include: options?.includes,
                exclude: options?.excludes
            }]
    };
}
