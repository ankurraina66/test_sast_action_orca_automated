export function buildToolCatalogue(
    localTools: McpTool[],
    remoteTools: McpTool[], ): ToolCapability[]{
    const localByName = new Map(
            localTools.map((tool) => [tool.name, tool]), );

    const remoteByName = new Map(
            remoteTools.map((tool) => [tool.name, tool]), );

    return Object.entries(TOOL_MAPPINGS).flatMap(
        ([capability, mapping]) => {
        const local = mapping.local
             ? localByName.get(mapping.local)
             : undefined;

        const remote = mapping.remote
             ? remoteByName.get(mapping.remote)
             : undefined;

        // Do not advertise capabilities absent from both providers.
        if (!local && !remote) {
            return [];
        }

        return [{
                capability,
                risk: mapping.risk,
                ...(local ? {
                    local
                }
                     : {}),
                ...(remote ? {
                    remote
                }
                     : {}),
            }
        ];
    }, );
}

export function resolveToolProvider(
    capability: ToolCapability,
    preferredProvider: AppScanProvider, ): {
    provider: AppScanProvider;
    tool: McpTool;
} {
    if (
        preferredProvider === "local" &&
        capability.local) {
        return {
            provider: "local",
            tool: capability.local,
        };
    }

    if (
        preferredProvider === "remote" &&
        capability.remote) {
        return {
            provider: "remote",
            tool: capability.remote,
        };
    }

    if (capability.local) {
        return {
            provider: "local",
            tool: capability.local,
        };
    }

    if (capability.remote) {
        return {
            provider: "remote",
            tool: capability.remote,
        };
    }

    throw new Error(
`No provider supports capability: ${capability.capability}`, );
}

export function normalizeCapabilityArguments(
    capability: string,
    provider: AppScanProvider,
    request: CapabilityRequest, ): Record < string, unknown > {
    const { capability: _capability, appId, scanId, issueId, ...common } = request;

    switch (capability) {
    case "scan-issues":
        return provider === "remote"
         ? {
            ...common,
            scope: "Scan",
            scopeId: scanId ?? request.scopeId,
        }
         : {
            ...common,
            scanId: scanId ?? request.scopeId,
        };

    case "issues":
        return provider === "local"
         ? {
            ...common,
            appId: appId ?? request.scopeId,
        }
         : {
            ...common,
            scope: request.scope ?? "Application",
            scopeId: appId ?? request.scopeId,
        };

    case "issue-details":
        return {
            issueId
        };

    default:
        return common;
    }
}
