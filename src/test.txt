
$ProjectRoot = "$HOME\ai-coding-assistants-claude-poc"

$SkillFiles = @(
    "$ProjectRoot\.claude\skills\appscan\SKILL.md",
    "$ProjectRoot\agent\skills\appscan\SKILL.md"
)

$SectionTitle = "## Response formatting and quality checks"

$Section = @'

## Response formatting and quality checks

- Write the final response in clear, grammatical, complete sentences. Before responding, check for truncated words, merged sentences, broken phrases, and incomplete bullet points.
- Use simple Markdown tables with concise headers. Ensure every row has the same number of cells and each value appears under the correct header.
- Do not split words across lines or columns intentionally. Avoid excessively wide tables; use a short list when a table would be difficult to read.
- Show complete issue IDs when they are relevant. Do not show partial IDs or incomplete ID-prefix lists as substitutes for full identifiers.
- Report only facts supported by AppScan tool responses. Clearly distinguish AppScan-returned facts from source-code observations, analysis, and inferences.
- If evidence, traces, descriptions, or remediation guidance are missing, state that clearly and briefly. Never invent missing content.
- If a field is returned as Unknown or unavailable, preserve its meaning accurately. Do not guess a replacement value.
- Keep limitations concise and specific to the requested task. Avoid repeating the same limitation in multiple sections.
- Do not claim that source code was inspected, a finding was validated, a file was changed, or a scan was run unless that action actually occurred.
- Before sending, verify that the response answers the request, contains no malformed text, and respects the requested scope.
- End with a follow-up offer only when it is relevant to the user's requested workflow. Make the offer a complete sentence.
'@

# Validate both files before changing either one.
foreach ($file in $SkillFiles) {
    if (-not (Test-Path -LiteralPath $file -PathType Leaf)) {
        throw "Skill file not found: $file"
    }
}

foreach ($file in $SkillFiles) {
    $content = [System.IO.File]::ReadAllText($file)

    if ($content.Contains($SectionTitle)) {
        Write-Host "Already present; no change: $file"
        continue
    }

    $updated = $content.TrimEnd() + "`r`n`r`n" + $Section.Trim() + "`r`n"
    $utf8NoBom = [System.Text.UTF8Encoding]::new($false)
    [System.IO.File]::WriteAllText($file, $updated, $utf8NoBom)

    Write-Host "Updated: $file"
}

# Confirm both copies contain the new section.
foreach ($file in $SkillFiles) {
    $content = [System.IO.File]::ReadAllText($file)
    [PSCustomObject]@{
        File = Split-Path $file -Parent
        FormattingRulesPresent = $content.Contains($SectionTitle)
    }
}
