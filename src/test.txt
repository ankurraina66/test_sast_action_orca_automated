import { describe, expect, it } from "vitest";
import { loadConfig } from "../../src/config/config.js";

describe("loadConfig", () => {
  it("defaults to remote mode and accepts a remote endpoint", () => {
    const config = loadConfig({
      APP_SCAN_REMOTE_ENDPOINT: "https://cloud.appscan.com/mcp",
    });

    expect(config.provider).toBe("remote");
    expect(config.remote?.endpoint).toBe(
      "https://cloud.appscan.com/mcp",
    );
    expect(config.requestTimeoutMs).toBe(30000);
  });

  it("accepts local mode with a configured server path", () => {
        const config = loadConfig({
          APP_SCAN_PROVIDER: "local",
          APP_SCAN_LOCAL_SERVER_PATH:
          "C:\\appscan-mcp\\build\\index.js",
        });
        expect(config.provider).toBe("local");
        expect(config.remote).toBeUndefined();
        expect(config.local?.command).toBe("node");
        expect(config.local?.serverPath).toBe(
    "C:\\appscan-mcp\\build\\index.js",
        );
        expect(config.local?.args).toEqual([
    "C:\\appscan-mcp\\build\\index.js",
        ]);
  });

  it("normalizes provider casing and whitespace", () => {
        const config = loadConfig({
          APP_SCAN_PROVIDER: " LOCAL ",
          APP_SCAN_LOCAL_SERVER_PATH:
      "C:\\appscan-mcp\\build\\index.js",
        });
        expect(config.provider).toBe("local");
  });

  it("rejects an unsupported provider", () => {
    expect(() =>
      loadConfig({
        APP_SCAN_PROVIDER: "unknown",
      }),
    ).toThrow('APP_SCAN_PROVIDER must be either "remote" or "local".');
  });

  it("requires an endpoint in remote mode", () => {
    expect(() => loadConfig({})).toThrow(
      "APP_SCAN_REMOTE_ENDPOINT is required for remote mode.",
    );
  });

  it("rejects an invalid remote endpoint", () => {
    expect(() =>
      loadConfig({
        APP_SCAN_REMOTE_ENDPOINT: "not-a-url",
      }),
    ).toThrow("APP_SCAN_REMOTE_ENDPOINT must be a valid URL.");
  });

  it("rejects unsupported URL protocols", () => {
    expect(() =>
      loadConfig({
        APP_SCAN_REMOTE_ENDPOINT: "ftp://example.com/mcp",
      }),
    ).toThrow("Remote endpoint must use HTTP or HTTPS.");
  });

  it("rejects invalid timeout values", () => {
    expect(() =>
      loadConfig({
        APP_SCAN_PROVIDER: "local",
        APP_SCAN_REQUEST_TIMEOUT_MS: "0",
      }),
    ).toThrow(
      "APP_SCAN_REQUEST_TIMEOUT_MS must be an integer between 1 and 300000.",
    );
  });

  it("accepts a configured remote API key without exposing it in errors", () => {
    const config = loadConfig({
      APP_SCAN_REMOTE_ENDPOINT: "https://cloud.appscan.com/mcp",
      APP_SCAN_API_KEY: "test-key",
    });

    expect(config.remote?.apiKey).toBe("test-key");
  });

  it("uses the configured local command and AppScan settings", () => {
    const config = loadConfig({
      APP_SCAN_PROVIDER: "local",
      APP_SCAN_LOCAL_SERVER_PATH: "C:\\appscan-mcp\\build\\index.js",
      APP_SCAN_LOCAL_COMMAND: "node",
      APPSCAN_API_BASE: "https://cloud.appscan.com",
      APPSCAN_KEY_ID: "test-key-id",
      APPSCAN_KEY_SECRET: "test-key-secret",
    });

    expect(config.local?.command).toBe("node");
    expect(config.local?.serverPath).toBe(
      "C:\\appscan-mcp\\build\\index.js",
    );
    expect(config.local?.apiBase).toBe("https://cloud.appscan.com");
    expect(config.local?.keyId).toBe("test-key-id");
    expect(config.local?.keySecret).toBe("test-key-secret");
  });

  it("requires a local server path in local mode", () => {
    expect(() =>
      loadConfig({
        APP_SCAN_PROVIDER: "local",
      }),
    ).toThrow("APP_SCAN_LOCAL_SERVER_PATH is required for local mode.");
  });
});
