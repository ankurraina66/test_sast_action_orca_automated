import type { AppScanMcpConnector, McpTool, McpToolCallResult,  } from "../mcp/connector.js";
import type { AppScanProvider } from "../config/config.js";

type ConnectorFactory = (
    provider: AppScanProvider, ) => AppScanMcpConnector;

export class AppScanService {
    private activeProvider: AppScanProvider;
    private operationQueue: Promise < void >  = Promise.resolve();

    constructor(
        private connector: AppScanMcpConnector,
        initialProvider: AppScanProvider,
        private readonly connectorFactory: ConnectorFactory, ) {
        this.activeProvider = initialProvider;
    }

    getProvider(): AppScanProvider {
        return this.activeProvider;
    }

    private async withLock < T > (
        operation: () => Promise < T > , ): Promise < T > {
        const previous = this.operationQueue;
        let release!: () => void;

        this.operationQueue = new Promise < void > ((resolve) => {
            release = resolve;
        });

        await previous;

        try {
            return await operation();
        } finally {
            release();
        }
    }

    async connect(): Promise < void > {
        await this.withLock(() => this.connector.connect());
    }

    async listTools(): Promise < McpTool[] > {
        return this.withLock(() => this.connector.listTools());
    }

    async callTool(
        name: string,
        args: Record < string, unknown >  = {}, ): Promise < McpToolCallResult > {
        return this.withLock(() => this.connector.callTool(name, args));
    }

    async switchProvider(
        provider: AppScanProvider, ): Promise < {
        provider: AppScanProvider;
        toolCount: number
    }
     > {
        return this.withLock(async() => {
            if (provider === this.activeProvider) {
                return {
                    provider: this.activeProvider,
                    toolCount: (await this.connector.listTools()).length,
                };
            }

            const previousConnector = this.connector;
            let nextConnector: AppScanMcpConnector | undefined;

            try {
                nextConnector = this.connectorFactory(provider);
                await nextConnector.connect();

                const tools = await nextConnector.listTools();

                if (tools.length === 0) {
                    throw new Error(
`The ${provider} AppScan provider returned no tools.`, );
                }

                // Activate only after connecting and validating the new provider.
                this.connector = nextConnector;
                this.activeProvider = provider;

                try {
                    await previousConnector.disconnect();
                } catch (error) {
                    // The new provider is active; report cleanup failure without
                    // incorrectly rolling back a successful switch.
                    console.error(
                        "The previous AppScan connector could not be cleanly disconnected.",
                        error instanceof Error ? error.message : "Unknown error", );
                }

                return {
                    provider,
                    toolCount: tools.length
                };
            } catch (error) {
                if (nextConnector && nextConnector !== this.connector) {
                    try {
                        await nextConnector.disconnect();
                    } catch {
                        // Preserve the original connection or validation error.
                    }
                }

                throw new Error(
`Could not switch to the ${provider} AppScan provider: ${
                    error instanceof Error ? error.message : "Unknown error"
}`, );
            }
        });
    }

    async findToolByNames(
        names: string[], ): Promise < McpTool | undefined > {
        return this.withLock(async() => {
            const tools = await this.connector.listTools();
            const normalizedNames = new Set(
                    names.map((name) => name.toLowerCase()), );

            return tools.find((tool) =>
                normalizedNames.has(tool.name.toLowerCase()), );
        });
    }

    async callFirstAvailableTool(
        names: string[],
        args: Record < string, unknown >  = {},
        expectedProvider ?  : AppScanProvider, ): Promise < McpToolCallResult > {
        return this.withLock(async() => {
            if (
                expectedProvider !== undefined &&
                this.activeProvider !== expectedProvider) {
                throw new Error(
`Provider mismatch: this operation was prepared for ${expectedProvider}, but the active provider is ${this.activeProvider}. The operation was not executed.`, );
            }

            // Discovery and execution use the same connector while holding the lock.
            const tools = await this.connector.listTools();
            const normalizedNames = new Set(
                    names.map((name) => name.toLowerCase()), );
            const tool = tools.find((item) =>
                    normalizedNames.has(item.name.toLowerCase()), );

            if (!tool) {
                throw new Error(
`The active ${this.activeProvider} AppScan provider does not support this operation. Expected one of: ${names.join(", ")}.`, );
            }

            return this.connector.callTool(tool.name, args);
        });
    }

    async disconnect(): Promise < void > {
        await this.withLock(() => this.connector.disconnect());
    }

    async findApplicationDiscoveryTool(): Promise < McpTool | undefined > {
        return this.withLock(async() => {
            const tools = await this.connector.listTools();

            return tools.find((tool) =>
                /^(get[_-]apps|list[_-]applications)$/i.test(tool.name), );
        });
    }

    async discoverApplications(
        args: Record < string, unknown >  = {}, ): Promise < McpToolCallResult > {
        return this.withLock(async() => {
            const tools = await this.connector.listTools();
            const tool = tools.find((item) =>
                    /^(get[_-]apps|list[_-]applications)$/i.test(item.name), );

            if (!tool) {
                throw new Error(
                    "The configured AppScan MCP server does not expose a recognized application discovery tool.", );
            }

            return this.connector.callTool(tool.name, args);
        });
    }
}
