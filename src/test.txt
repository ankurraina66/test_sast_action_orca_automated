# Enter credentials securely
$keyIdSecure = Read-Host "Enter AppScan Key ID" -AsSecureString
$keySecretSecure = Read-Host "Enter AppScan Key Secret" -AsSecureString

# Convert securely entered values temporarily for MCP configuration
$keyIdPtr = [Runtime.InteropServices.Marshal]::SecureStringToBSTR($keyIdSecure)
$keySecretPtr = [Runtime.InteropServices.Marshal]::SecureStringToBSTR($keySecretSecure)

try {
    $keyId = [Runtime.InteropServices.Marshal]::PtrToStringBSTR($keyIdPtr)
    $keySecret = [Runtime.InteropServices.Marshal]::PtrToStringBSTR($keySecretPtr)

    $config = @{
        type = "stdio"
        command = "C:\Users\ankur.raina\AppData\Roaming\fnm\node-versions\v22.23.4\installation\node.exe"
        args = @(
            "C:\Users\ankur.raina\ai-coding-assistants-claude-poc\dist\claude-mcp-server.js"
        )
        env = @{
            APP_SCAN_PROVIDER = "local"
            APP_SCAN_LOCAL_COMMAND = "C:\Users\ankur.raina\AppData\Roaming\fnm\node-versions\v22.23.4\installation\node.exe"
            APP_SCAN_LOCAL_ARGS = '["C:\\Users\\ankur.raina\\AppData\\Roaming\\fnm\\node-versions\\v22.23.4\\installation\\node_modules\\mcp-appscan-server\\build\\index.js"]'
            APP_SCAN_REQUEST_TIMEOUT_MS = "30000"
            APP_SCAN_REMOTE_ENDPOINT = "https://cloud.appscan.com/mcp"
            APP_SCAN_API_KEY = "${keyId}:${keySecret}"
            APPSCAN_API_BASE = "https://cloud.appscan.com"
            APPSCAN_KEY_ID = $keyId
            APPSCAN_KEY_SECRET = $keySecret
        }
    }

    $json = $config | ConvertTo-Json -Depth 5 -Compress

    & claude mcp add-json --scope user hcl-appscan-ai-coding-assistants $json
}
finally {
    [Runtime.InteropServices.Marshal]::ZeroFreeBSTR($keyIdPtr)
    [Runtime.InteropServices.Marshal]::ZeroFreeBSTR($keySecretPtr)

    Remove-Variable keyId, keySecret, json, config -ErrorAction SilentlyContinue
    Remove-Variable keyIdSecure, keySecretSecure -ErrorAction SilentlyContinue
}
