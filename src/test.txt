package com.hcl.appscan.intellij.plugin.ui;

import com.hcl.appscan.intellij.plugin.auth.asoc.LoginUtility;
import com.hcl.appscan.sdk.auth.IAuthenticationProvider;
import com.intellij.openapi.actionSystem.AnAction;
import com.intellij.openapi.project.Project;
import com.intellij.openapi.ui.SimpleToolWindowPanel;
import com.intellij.ui.components.JBScrollPane;
import javax.swing.*;

public class IssueDetailsToolWindow {
    private HowToFixPanel howToFixPanel;
    private DataFlowPanel dataFlowPanel;
    private IAuthenticationProvider authenticationHandler;

    public IssueDetailsToolWindow(Project project) {
        authenticationHandler = LoginUtility.getASoCHandler();
    }

    public SimpleToolWindowPanel createHowToFixToolWindowContent(final Project project) {
        howToFixPanel = new HowToFixPanel(project);
        SimpleToolWindowPanel mainPanel = new SimpleToolWindowPanel(true, true);
        JPanel fixGroupPanelComponent = howToFixPanel.getComponent();
        JBScrollPane scrollPane = new JBScrollPane(fixGroupPanelComponent, ScrollPaneConstants.VERTICAL_SCROLLBAR_AS_NEEDED, ScrollPaneConstants.HORIZONTAL_SCROLLBAR_AS_NEEDED);
        scrollPane.setViewportView(fixGroupPanelComponent);
        mainPanel.add(scrollPane);
        return mainPanel;
    }

    public SimpleToolWindowPanel createDataFlowToolWindowContent(final Project project) {
        dataFlowPanel = new DataFlowPanel(project);
        SimpleToolWindowPanel mainPanel = new SimpleToolWindowPanel(true, true);
        JPanel fixGroupPanelComponent = dataFlowPanel.getComponent();
        JBScrollPane scrollPane = new JBScrollPane(fixGroupPanelComponent, ScrollPaneConstants.VERTICAL_SCROLLBAR_AS_NEEDED, ScrollPaneConstants.HORIZONTAL_SCROLLBAR_AS_NEEDED);
        scrollPane.setViewportView(fixGroupPanelComponent);
        mainPanel.add(scrollPane);
        return mainPanel;
    }
}

package com.hcl.appscan.intellij.plugin.ui;

import com.hcl.appscan.intellij.plugin.MessagesBundle;
import com.hcl.appscan.intellij.plugin.util.AndroidStudioUtil;
import com.hcl.appscan.intellij.plugin.util.UIComponentsUtil;
import com.intellij.openapi.project.DumbAware;
import com.intellij.openapi.project.Project;
import com.intellij.openapi.ui.SimpleToolWindowPanel;
import com.intellij.openapi.wm.ToolWindow;
import com.intellij.openapi.wm.ToolWindowFactory;
import com.intellij.openapi.wm.impl.content.ToolWindowContentUi;
import com.intellij.ui.content.Content;
import com.intellij.ui.content.ContentFactory;
import com.intellij.ui.content.ContentManager;
import org.jetbrains.annotations.NotNull;

public class IssueDetailsToolWindowFactory implements ToolWindowFactory, DumbAware {

	@Override
	public void createToolWindowContent(@NotNull Project project, @NotNull ToolWindow toolWindow) {
		if (!AndroidStudioUtil.isAndroidStudio()) {
			UIComponentsUtil.setDetailsToolwindow(toolWindow);
			IssueDetailsToolWindow issueDetailsToolWindow = new IssueDetailsToolWindow(project);

			toolWindow.getComponent().putClientProperty(ToolWindowContentUi.HIDE_ID_LABEL, "true");
			ContentManager contentManager = toolWindow.getContentManager();
			SimpleToolWindowPanel howToFixToolWindowContent = issueDetailsToolWindow.createHowToFixToolWindowContent(project);
			Content howToFixGroupContent = ContentFactory.SERVICE.getInstance().createContent(howToFixToolWindowContent, MessagesBundle.message("toolwindow.issuedetails.howtofix.title"), false);
			contentManager.addContent(howToFixGroupContent);

			SimpleToolWindowPanel dataFlowToolWindowContent = issueDetailsToolWindow.createDataFlowToolWindowContent(project);
			Content dataFlowContent = ContentFactory.SERVICE.getInstance().createContent(dataFlowToolWindowContent, MessagesBundle.message("toolwindow.issuedetails.dataflow.title"), false);
			contentManager.addContent(dataFlowContent);

			contentManager.setSelectedContent(howToFixGroupContent);
			toolWindow.hide();
		}
	}
}
