function jsonSchemaToZod(schema: unknown): z.ZodTypeAny {
    if (!schema || typeof schema !== "object") {
        return z.unknown();
    }

    const definition = schema as Record < string,
    unknown > ;

    if (definition.const !== undefined) {
        return z.literal(
            definition.const as string | number | boolean, );
    }

    if (Array.isArray(definition.enum)) {
        const values = definition.enum;

        if (
            values.length > 0 &&
            values.every((value) => typeof value === "string")) {
            return z.enum(values as[string, ...string[]]);
        }

        if (values.length === 1) {
            return z.literal(values[0]);
        }

        if (values.length > 1) {
            const literals = values.map((value) => z.literal(value));

            return z.union(
                literals as unknown as[
                    z.ZodTypeAny,
                    z.ZodTypeAny,
                    ...z.ZodTypeAny[],
                ], );
        }
    }

    if (
        Array.isArray(definition.anyOf) ||
        Array.isArray(definition.oneOf)) {
        const variants = (definition.anyOf ?? definition.oneOf)as unknown[];

        if (variants.length === 1) {
            return jsonSchemaToZod(variants[0]);
        }

        if (variants.length >= 2) {
            const schemas = variants.map(jsonSchemaToZod);

            return z.union(
                schemas as unknown as[
                    z.ZodTypeAny,
                    z.ZodTypeAny,
                    ...z.ZodTypeAny[],
                ], );
        }
    }

    const type = definition.type;
    let result: z.ZodTypeAny;

    if (type === "string") {
        let value = z.string();

        if (typeof definition.minLength === "number") {
            value = value.min(definition.minLength);
        }

        if (typeof definition.maxLength === "number") {
            value = value.max(definition.maxLength);
        }

        if (typeof definition.pattern === "string") {
            value = value.regex(new RegExp(definition.pattern));
        }

        result = value;
    } else if (type === "integer") {
        let value = z.number().int();

        if (typeof definition.minimum === "number") {
            value = value.min(definition.minimum);
        }

        if (typeof definition.maximum === "number") {
            value = value.max(definition.maximum);
        }

        result = value;
    } else if (type === "number") {
        let value = z.number();

        if (typeof definition.minimum === "number") {
            value = value.min(definition.minimum);
        }

        if (typeof definition.maximum === "number") {
            value = value.max(definition.maximum);
        }

        result = value;
    } else if (type === "boolean") {
        result = z.boolean();
    } else if (type === "array") {
        result = z.array(
                jsonSchemaToZod(definition.items), );
    } else if (type === "object" || definition.properties) {
        const properties = (
            definition.properties ?? {})as Record < string,
        unknown > ;

        const required = new Set(
                Array.isArray(definition.required)
                 ? definition.required.filter(
                    (item): item is string => typeof item === "string", )
                 : [], );

        const shape: Record < string,
        z.ZodTypeAny >  = {};

        for (const [key, propertySchema] of Object.entries(properties)) {
            const property = jsonSchemaToZod(propertySchema);

            shape[key] = required.has(key)
                 ? property
                 : property.optional();
        }

        result = z.object(shape);
    } else {
        result = z.unknown();
    }

    if (definition.nullable === true) {
        result = result.nullable();
    }

    if (definition.default !== undefined) {
        result = result.default(definition.default);
    }

    return result;
}

Returns promptly; use get-scan-execution-status to retrieve its eventual result. Never call this tool merely because a preparation token exists.",
        inputSchema: {
            confirmationToken: z.string().uuid(),

        if (!prepared) {
            return messageResult(
                "No pending scan matches this token. Prepare the request again and obtain user confirmation.",
                true, );
        }

        if (prepared.provider !== "local") {
            return messageResult(
                "Scan execution is supported only by the Local AppScan MCP backend. The scan was not started.",
                true, );
        }

        if (executingScanTokens.has(confirmationToken)) {
            return messageResult(
                "This confirmation token is already being processed. Do not submit it again.",
                true, );
        }

        if (prepared.provider !== service.getProvider()) {
            return messageResult(
`This scan was prepared for the ${prepared.provider} provider, but the active provider is ${service.getProvider()}. The scan was not started. Prepare the request again and obtain explicit confirmation.`,
                true, );
        }

        // Claim and consume the approval token before any work starts, so a
        // token can never start a second scan.
        executingScanTokens.add(confirmationToken);
        pendingScans.delete(confirmationToken);

        const startedAt = new Date().toISOString();

        scanExecutions.set(confirmationToken, {
            status: "running",
            startedAt,
            phase: "creating",
        });

        const redact = createRedactor();

        // One lifecycle per token: create once, poll until a confirmed outcome,
        // then collect results. It is never retried or re-submitted.
        const lifecycle = (async() => {
            try {
                const result = await runScanLifecycle(
                        prepared, {
                        callTool: (names, args) =>
                        service.callFirstAvailableToolIfSupported(
                            names,
                            args,
                            "local", ),
                        redact,
                        onProgress: (progress) => {
                            scanExecutions.set(confirmationToken, {
                                status: "running",
                                startedAt,
                                phase: progress.phase,
                                ...(progress.scanId !== undefined
                                     ? {
                                    scanId: progress.scanId
                                }
                                     : {}),
                                ...(progress.lastKnownStatus !== undefined
                                     ? {
                                    lastKnownStatus: progress.lastKnownStatus
                                }
                                     : {}),
                            });
                        },
                    }, {
                        ...lifecycleConfig,
                        ...DEFAULT_LIFECYCLE_OPTIONS,
                    }, );

                scanExecutions.set(confirmationToken, {
                    status: result.outcome,
                    startedAt,
                    completedAt: result.endedAt,
                    result,
                });

                return result;
            } finally {
                executingScanTokens.delete(confirmationToken);
            }
        })().catch((error: unknown) => {
            const message = redact(
                    error instanceof Error ? error.message : "Unknown error", );

            console.error(
`[AppScan scan execution] token=${confirmationToken} ended unexpectedly: ${message}`, );

            scanExecutions.set(confirmationToken, {
                status: "unconfirmed",
                startedAt,
                completedAt: new Date().toISOString(),
                error: message,
            });

            return undefined;
        });

        if (!waitForCompletion) {
            return messageResult(
                [
                    "Scan execution was accepted by the connector and is running in the background.",
`Confirmation token: ${confirmationToken}`,
                    "Use get-scan-execution-status with this token to retrieve the final result.",
                    "Do not submit another scan while the status is running or uncertain.",
                ].join("\n"), );
        }

        const outcome = await lifecycle;

        if (!outcome) {
            return messageResult(
                [
                    "Outcome: UNCONFIRMED",
                    "The scan lifecycle ended unexpectedly. Do not submit another scan.",
`Confirmation token: ${confirmationToken}`,
                    "Use get-scan-execution-status with this token to see what is known.",
                ].join("\n"),
                true, );
        }

        return messageResult(
            formatScanLifecycleResult(outcome),
            outcome.outcome !== "completed", );
    }, );

    server.registerTool(
        "get-scan-execution-status", {
        description:
        "Retrieve the status and final result of a previously approved scan execution. This tool never starts a scan.",
        inputSchema: {
            confirmationToken: z.string().uuid(),
        },
    },
        async({
            confirmationToken
        }) => {
        const execution = scanExecutions.get(confirmationToken);

        if (!execution) {
            return messageResult(
                "No execution record exists for this token. Do not retry the scan automatically; verify the AppScan backend first.",
                true, );
        }

        if (execution.status === "running") {
            return messageResult(
                JSON.stringify({
                    status: execution.status,
                    startedAt: execution.startedAt,
                    ...(execution.phase !== undefined
                         ? {
                        phase: execution.phase
                    }
                         : {}),
                    ...(execution.scanId !== undefined
                         ? {
                        scanId: execution.scanId
                    }
                         : {}),
                    ...(execution.lastKnownStatus !== undefined
                         ? {
                        lastKnownStatus: execution.lastKnownStatus
                    }
                         : {}),
                    confirmationToken,
                },
                    null,
                    2, ), );
        }

        const finished = execution.status === "completed";

        return messageResult(
            JSON.stringify({
                status: execution.status,
                startedAt: execution.startedAt,
                completedAt: execution.completedAt,
                confirmationToken,
                ...(execution.result !== undefined
                     ? {
                    result: execution.result
                }
                     : {}),
                ...(execution.error !== undefined
                     ? {
                    error: execution.error
                }
                     : {}),
                ...(finished
                     ? {}
                     : {
                    warning:
                    "The final backend outcome may require verification. Do not automatically start another scan.",
                }),
            },
                null,
                2, ),
            !finished, );
    }, );

    const transport = new StdioServerTransport();

    try {
        await server.connect(transport);
        console.error("HCL AppScan Connector for Claude is running.");
    } catch (error) {
        await service.disconnect();
        throw error;
    }

    const shutdown = async(): Promise < void >  => {
        try {
            await server.close();
        } finally {
            await service.disconnect();
        }
    };

    process.once("SIGINT", () => {
        void shutdown().catch((error: unknown) => {
            console.error("Shutdown failed:", error);
            process.exitCode = 1;
        });
    });

    process.once("SIGTERM", () => {
        void shutdown().catch((error: unknown) => {
            console.error("Shutdown failed:", error);
            process.exitCode = 1;
        });
    });
}

main().catch((error: unknown) => {
    console.error(
        "Failed to start HCL AppScan Connector for Claude:",
        error instanceof Error ? error.message : "Unknown error", );
    process.exitCode = 1;
});
