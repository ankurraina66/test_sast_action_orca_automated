
it("lists tools from an inactive provider without changing the active provider", async () => {
    const active = createMockConnector(["get-apps"]);
    const inactive = createMockConnector(["get_apps", "get_scans"]);
    const factory = vi.fn(() => inactive);

    const service = createService(active, "local", factory);

    await expect(
        service.listProviderTools("remote"),
    ).resolves.toMatchObject([
        { name: "get_apps" },
        { name: "get_scans" },
    ]);

    expect(factory).toHaveBeenCalledWith("remote");
    expect(inactive.connect).toHaveBeenCalledOnce();
    expect(inactive.listTools).toHaveBeenCalledOnce();
    expect(inactive.disconnect).toHaveBeenCalledOnce();

    // Active provider must remain unchanged.
    expect(service.getProvider()).toBe("local");
    expect(active.disconnect).not.toHaveBeenCalled();
});
