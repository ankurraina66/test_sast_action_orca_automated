import { loadConfig } from "../config/config.js";
import type { AppScanProvider } from "../config/config.js";
import { createConnector } from "../mcp/connector-factory.js";
import { AppScanService } from "./appscan-service.js";

export function createAppScanService(): AppScanService {
  const startupEnvironment = { ...process.env };

  console.error("[AppScan MCP] Startup configuration:", {
    provider: startupEnvironment.APP_SCAN_PROVIDER ?? "remote (default)",
    remoteEndpointConfigured: Boolean(
      startupEnvironment.APP_SCAN_REMOTE_ENDPOINT?.trim(),
    ),
    remoteApiKeyConfigured: Boolean(
      startupEnvironment.APP_SCAN_API_KEY?.trim(),
    ),
  });

  const config = loadConfig(startupEnvironment);
  const connector = createConnector(config);

  return new AppScanService(
    connector,
    config.provider,
    (provider: AppScanProvider) => {
      console.error("[AppScan MCP] Provider switch requested:", {
        provider,
        remoteEndpointConfigured: Boolean(
          startupEnvironment.APP_SCAN_REMOTE_ENDPOINT?.trim(),
        ),
        remoteApiKeyConfigured: Boolean(
          startupEnvironment.APP_SCAN_API_KEY?.trim(),
        ),
      });

      return createConnector(
        loadConfig({
          ...startupEnvironment,
          APP_SCAN_PROVIDER: provider,
        }),
      );
    },
  );
}
