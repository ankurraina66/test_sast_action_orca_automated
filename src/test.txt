import { Client } from "@modelcontextprotocol/sdk/client/index.js";
import { StreamableHTTPClientTransport } from "@modelcontextprotocol/sdk/client/streamableHttp.js";
import type { Tool } from "@modelcontextprotocol/sdk/types.js";

import type {
AppScanMcpConnector,
McpTool,
McpToolCallResult,
} from "./connector.js";

export interface RemoteHttpConnectorOptions {
endpoint: string;
apiKey?: string;
requestTimeoutMs?: number;
}

export class RemoteHttpConnector implements AppScanMcpConnector {
private readonly client: Client;
private readonly transport: StreamableHTTPClientTransport;
private connected = false;

constructor(options: RemoteHttpConnectorOptions) {
this.client = new Client({
name: "hcl-appscan-connector-for-claude",
version: "0.1.0",
});

const headers: Record<string, string> = {};

if (options.apiKey) {
  headers["X-API-KEY"] = options.apiKey;
}

this.transport = new StreamableHTTPClientTransport(
  new URL(options.endpoint),
  {
    requestInit: {
      headers,
      signal: options.requestTimeoutMs
        ? AbortSignal.timeout(options.requestTimeoutMs)
        : undefined,
    },
  },
);

}

async connect(): Promise<void> {
if (this.connected) {
return;
}

await this.client.connect(this.transport);
this.connected = true;

}

async listTools(): Promise<McpTool[]> {
this.ensureConnected();

const result = await this.client.listTools();

return result.tools.map((tool: Tool) => ({
  name: tool.name,
  description: tool.description,
  inputSchema: tool.inputSchema as Record<string, unknown>,
}));

}

async callTool(
name: string,
args: Record<string, unknown>,
): Promise<McpToolCallResult> {
this.ensureConnected();

const result: unknown = await this.client.callTool({
  name,
  arguments: args,
});

if (
  typeof result !== "object" ||
  result === null ||
  !("content" in result) ||
  !Array.isArray(result.content)
) {
  throw new Error(
    `MCP tool "${name}" returned an unsupported result format.`,
  );
}

const content: McpToolCallResult["content"] = result.content.map(
  (item: unknown) => {
    if (typeof item !== "object" || item === null || !("type" in item)) {
      throw new Error(
        `MCP tool "${name}" returned an invalid content item.`,
      );
    }

    if (typeof item.type !== "string") {
      throw new Error(
        `MCP tool "${name}" returned a content item with an invalid type.`,
      );
    }

    if ("text" in item && typeof item.text === "string") {
      return { type: item.type, text: item.text };
    }

    return { type: item.type };
  },
);

return {
  content,
  isError:
    "isError" in result && typeof result.isError === "boolean"
      ? result.isError
      : undefined,
};

}

async disconnect(): Promise<void> {
if (!this.connected) {
return;
}

try {
  await this.client.close();
} finally {
  this.connected = false;
}

}

private ensureConnected(): void {
if (!this.connected) {
throw new Error("Remote AppScan MCP connector is not connected.");
}
}
}

import { beforeEach, describe, expect, it, vi } from "vitest";

const { mockClient, mockTransport } = vi.hoisted(() => ({
mockClient: {
connect: vi.fn(),
listTools: vi.fn(),
callTool: vi.fn(),
close: vi.fn(),
},
mockTransport: vi.fn(),
}));

vi.mock("@modelcontextprotocol/sdk/client/index.js", () => ({
Client: vi.fn(function () {
return mockClient;
}),
}));

vi.mock(
"@modelcontextprotocol/sdk/client/streamableHttp.js",
() => ({
StreamableHTTPClientTransport: mockTransport,
}),
);

import { RemoteHttpConnector } from "../../src/mcp/remote-http-connector.js";

describe("RemoteHttpConnector", () => {
let connector: RemoteHttpConnector;

beforeEach(() => {
vi.clearAllMocks();

mockClient.connect.mockResolvedValue(undefined);
mockClient.listTools.mockResolvedValue({ tools: [] });
mockClient.callTool.mockResolvedValue({
  content: [{ type: "text", text: "Success" }],
  isError: false,
});
mockClient.close.mockResolvedValue(undefined);

mockTransport.mockImplementation(function () {
  return {};
});

connector = new RemoteHttpConnector({
  endpoint: "https://cloud.appscan.com/mcp",
  apiKey: "test-key",
});

});

it("connects to the remote MCP server", async () => {
await connector.connect();

expect(mockClient.connect).toHaveBeenCalledTimes(1);

});

it("does not connect twice when already connected", async () => {
await connector.connect();
await connector.connect();

expect(mockClient.connect).toHaveBeenCalledTimes(1);

});

it("lists tools exposed by the MCP server", async () => {
mockClient.listTools.mockResolvedValue({
tools: [
{
name: "get_apps",
description: "List applications",
inputSchema: { type: "object", properties: {} },
},
],
});

await connector.connect();

await expect(connector.listTools()).resolves.toEqual([
  {
    name: "get_apps",
    description: "List applications",
    inputSchema: { type: "object", properties: {} },
  },
]);

});

it("rejects tool listing before connecting", async () => {
await expect(connector.listTools()).rejects.toThrow(
"Remote AppScan MCP connector is not connected.",
);
});

it("calls a tool and normalizes its result", async () => {
await connector.connect();

await expect(
  connector.callTool("get_apps", { top: 5 }),
).resolves.toEqual({
  content: [{ type: "text", text: "Success" }],
  isError: false,
});

expect(mockClient.callTool).toHaveBeenCalledWith({
  name: "get_apps",
  arguments: { top: 5 },
});

});

it("rejects a tool call before connecting", async () => {
await expect(
connector.callTool("get_apps", {}),
).rejects.toThrow(
"Remote AppScan MCP connector is not connected.",
);
});

it("rejects an unsupported tool result format", async () => {
await connector.connect();
mockClient.callTool.mockResolvedValue({ unexpected: true });

await expect(
  connector.callTool("get_apps", {}),
).rejects.toThrow(
  'MCP tool "get_apps" returned an unsupported result format.',
);

});

it("disconnects the client", async () => {
await connector.connect();
await connector.disconnect();

expect(mockClient.close).toHaveBeenCalledTimes(1);

});

it("passes the API key in the X-API-KEY header", () => {
  expect(mockTransport).toHaveBeenCalledWith(
    new URL("https://cloud.appscan.com/mcp"),
    expect.objectContaining({
      requestInit: expect.objectContaining({
        headers: {
          "X-API-KEY": "test-key",
        },
      }),
    }),
  );
});

it("configures the request timeout when provided", () => {
  new RemoteHttpConnector({
    endpoint: "https://cloud.appscan.com/mcp",
    apiKey: "test-key",
    requestTimeoutMs: 5000,
  });

  expect(mockTransport).toHaveBeenCalledWith(
    new URL("https://cloud.appscan.com/mcp"),
    expect.objectContaining({
      requestInit: expect.objectContaining({
        signal: expect.any(AbortSignal),
      }),
    }),
  );
});

it("does nothing when disconnecting before connecting", async () => {
await connector.disconnect();

expect(mockClient.close).not.toHaveBeenCalled();

});
});
