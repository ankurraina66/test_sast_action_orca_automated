import { loadConfig } from "../config/config.js";
import type { AppScanProvider } from "../config/config.js";
import { createConnector } from "../mcp/connector-factory.js";
import { AppScanService } from "./appscan-service.js";

export function createAppScanService(): AppScanService {
  const config = loadConfig();
  const connector = createConnector(config);

  return new AppScanService(
    connector,
    config.provider,
    (provider: AppScanProvider) =>
      createConnector(
        loadConfig({
          ...process.env,
          APP_SCAN_PROVIDER: provider,
        }),
      ),
  );
}
