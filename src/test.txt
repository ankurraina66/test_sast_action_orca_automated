server.registerTool(
    "execute-appscan-read",
    {
        description:
            "Execute a read-only AppScan capability using the unified Local/Remote routing service. Local is preferred; Remote is used when Local does not support the capability. Does not switch the active provider.",
        inputSchema: {
            capability: z.string().min(1),
            arguments: z.record(z.string(), z.unknown()).default({}),
        },
    },
    async ({ capability, arguments: toolArguments }) => {
        try {
            const result = await service.callCapability(
                capability,
                toolArguments,
            );

            return textResult(result);
        } catch (error) {
            return messageResult(
                error instanceof Error
                    ? error.message
                    : "AppScan capability execution failed.",
                true,
            );
        }
    },
);
