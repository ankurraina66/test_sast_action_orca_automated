
import { describe, expect, it } from "vitest";
import {
    buildToolCatalogue,
    resolveToolProvider,
} from "../../src/services/tool-catalogue.js";
import type { McpTool } from "../../src/mcp/connector.js";

function tool(name: string): McpTool {
    return {
        name,
        description: `${name} description`,
        inputSchema: {
            type: "object",
            properties: {},
        },
    };
}

describe("tool catalogue", () => {
    it("merges Local and Remote capabilities", () => {
        const catalogue = buildToolCatalogue(
            [tool("get-apps"), tool("create-scan")],
            [tool("get_apps"), tool("get_licenses")],
        );

        expect(catalogue).toHaveLength(3);

        const applications = catalogue.find(
            (item) => item.capability === "applications",
        );

        expect(applications?.local?.name).toBe("get-apps");
        expect(applications?.remote?.name).toBe("get_apps");
    });

    it("preserves the original backend schemas", () => {
        const local = tool("get-apps");
        local.inputSchema = {
            type: "object",
            properties: {
                top: { type: "number" },
            },
        };

        const catalogue = buildToolCatalogue(
            [local],
            [],
        );

        expect(
            catalogue[0].local?.inputSchema,
        ).toBe(local.inputSchema);
    });

    it("prefers Local when both providers support a capability", () => {
        const catalogue = buildToolCatalogue(
            [tool("get-apps")],
            [tool("get_apps")],
        );

        const result = resolveToolProvider(
            catalogue[0],
            "local",
        );

        expect(result.provider).toBe("local");
        expect(result.tool.name).toBe("get-apps");
    });

    it("uses Remote when Local does not support a capability", () => {
        const catalogue = buildToolCatalogue(
            [],
            [tool("get_licenses")],
        );

        const result = resolveToolProvider(
            catalogue[0],
            "local",
        );

        expect(result.provider).toBe("remote");
        expect(result.tool.name).toBe("get_licenses");
    });

    it("classifies write operations correctly", () => {
        const catalogue = buildToolCatalogue(
            [tool("update-issues"), tool("create-scan")],
            [tool("update_issues")],
        );

        expect(
            catalogue.find(
                (item) => item.capability === "update-issues",
            )?.risk,
        ).toBe("write");

        expect(
            catalogue.find(
                (item) => item.capability === "create-scan",
            )?.risk,
        ).toBe("write");
    });
});
