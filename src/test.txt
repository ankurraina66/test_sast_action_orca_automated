appscan-service.ts

import type { AppScanMcpConnector, McpTool, McpToolCallResult,  } from "../mcp/connector.js";
import type { AppScanProvider } from "../config/config.js";
import { Ajv } from "ajv";
import type { ErrorObject } from "ajv";
import { buildToolCatalogue, resolveToolProvider, normalizeCapabilityArguments,  } from "./tool-catalogue.js";
import type { CapabilityRequest } from "./tool-catalogue.js";

type ConnectorFactory = (
    provider: AppScanProvider, ) => AppScanMcpConnector;

export class AppScanService {
    private activeProvider: AppScanProvider;
    private operationQueue: Promise < void >  = Promise.resolve();

    private readonly ajv = new Ajv({
        allErrors: true,
        strict: false,
    });

    constructor(
        private connector: AppScanMcpConnector,
        initialProvider: AppScanProvider,
        private readonly connectorFactory: ConnectorFactory, ) {
        this.activeProvider = initialProvider;
    }

    getProvider(): AppScanProvider {
        return this.activeProvider;
    }

    private async withLock < T > (
        operation: () => Promise < T > , ): Promise < T > {
        const previous = this.operationQueue;
        let release!: () => void;

        this.operationQueue = new Promise < void > ((resolve) => {
            release = resolve;
        });

        await previous;

        try {
            return await operation();
        } finally {
            release();
        }
    }

    async connect(): Promise < void > {
        await this.withLock(() => this.connector.connect());
    }

    async listTools(): Promise < McpTool[] > {
        return this.withLock(() => this.connector.listTools());
    }

    /**
     * Discover tools from either provider without changing the active provider.
     * An inactive provider is connected temporarily and always disconnected.
     */
    async listProviderTools(
        provider: AppScanProvider, ): Promise < McpTool[] > {
        return this.withLock(async() => {
            if (provider === this.activeProvider) {
                return this.connector.listTools();
            }

            const temporaryConnector = this.connectorFactory(provider);

            try {
                await temporaryConnector.connect();
                return await temporaryConnector.listTools();
            } finally {
                await temporaryConnector.disconnect();
            }
        });
    }

    /**
     * Discover both backend tool catalogs without switching the active provider.
     */
    async compareProviderTools(): Promise < {
        local: McpTool[];
        remote: McpTool[];
    }
     > {
        return this.withLock(async() => {
            const getTools = async(
                    provider: AppScanProvider, ): Promise < McpTool[] >  => {
                if (provider === this.activeProvider) {
                    return this.connector.listTools();
                }

                const temporaryConnector = this.connectorFactory(provider);

                try {
                    await temporaryConnector.connect();
                    return await temporaryConnector.listTools();
                } finally {
                    await temporaryConnector.disconnect();
                }
            };

            const local = await getTools("local");
            const remote = await getTools("remote");

            return {
                local,
                remote,
            };
        });
    }

    /**
     * Execute a tool against a specific provider without changing
     * the currently active provider.
     */
    async callProviderTool(
        provider: AppScanProvider,
        toolName: string,
        args: Record < string, unknown >  = {}, ): Promise < McpToolCallResult > {
        return this.withLock(async() => {
            // Reuse the active connector when possible.
            if (provider === this.activeProvider) {
                return this.connector.callTool(toolName, args);
            }

            // Connect temporarily to the requested inactive provider.
            const temporaryConnector = this.connectorFactory(provider);

            try {
                await temporaryConnector.connect();

                return await temporaryConnector.callTool(
                    toolName,
                    args, );
            } finally {
                await temporaryConnector.disconnect();
            }
        });
    }

    /**
     * Execute a read-only unified capability.
     * Discovers Local and Remote independently, so one unavailable
     * provider does not prevent execution through the other.
     * Arguments are normalized and validated against the selected
     * provider's actual MCP tool schema.
     */
    async callCapability(
        capability: string,
        args: Record < string, unknown >  = {}, ): Promise < McpToolCallResult > {
        // Discover both providers independently.
        const [localResult, remoteResult] = await Promise.allSettled([
                    this.listProviderTools("local"),
                    this.listProviderTools("remote"),
                ]);

        const local =
            localResult.status === "fulfilled"
             ? localResult.value
             : [];

        const remote =
            remoteResult.status === "fulfilled"
             ? remoteResult.value
             : [];

        // Neither provider could be discovered.
        if (
            localResult.status === "rejected" &&
            remoteResult.status === "rejected") {
            throw new Error(
                "Unable to discover AppScan tools from either provider. " +
                "Check the Local and Remote provider configurations and connections.", );
        }

        // Build the catalogue from whichever providers are available.
        const catalogue = buildToolCatalogue(local, remote);

        const selected = catalogue.find(
                (item) => item.capability === capability, );

        if (!selected) {
            const failures: string[] = [];

            if (localResult.status === "rejected") {
                failures.push("Local provider discovery failed.");
            }

            if (remoteResult.status === "rejected") {
                failures.push("Remote provider discovery failed.");
            }

            throw new Error(
`Unsupported or unavailable AppScan capability: ${capability}. ` +
                (failures.length > 0 ? failures.join(" ") : ""), );
        }

        if (selected.risk !== "read") {
            throw new Error(
`Capability '${capability}' is not read-only. Explicit approval is required.`, );
        }

        // Prefer Local when available; otherwise use Remote.
        const resolved = resolveToolProvider(selected, "local");

        // Normalize unified arguments into provider-specific arguments.
        const request: CapabilityRequest = {
            ...args,
            capability,
        };

        const normalizedArgs = normalizeCapabilityArguments(
                capability,
                resolved.provider,
                request, );

        // Validate against the selected backend's actual MCP schema.
        const validate = this.ajv.compile(
                resolved.tool.inputSchema, );

        if (!validate(normalizedArgs)) {
            const errors = (validate.errors ?? [])
            .map(
                (error: ErrorObject) =>
`${error.instancePath || "/"}: ${error.message ?? "Invalid value"}`, )
            .join("; ");

            throw new Error(
`Invalid arguments for ${resolved.tool.name}: ${errors}`, );
        }

        // Execute without changing the active provider.
        return this.callProviderTool(
            resolved.provider,
            resolved.tool.name,
            normalizedArgs, );
    }
	
	/**
     * Validate a write capability and return the exact execution preview.
     * No AppScan operation is executed here.
     */
    async prepareApprovedCapability(
        capability: string,
        args: Record<string, unknown> = {},
    ): Promise<{
        provider: AppScanProvider;
        toolName: string;
        normalizedArgs: Record<string, unknown>;
    }> {
        const [localResult, remoteResult] = await Promise.allSettled([
            this.listProviderTools("local"),
            this.listProviderTools("remote"),
        ]);

        const local =
            localResult.status === "fulfilled" ? localResult.value : [];
        const remote =
            remoteResult.status === "fulfilled" ? remoteResult.value : [];

        if (
            localResult.status === "rejected" &&
            remoteResult.status === "rejected"
        ) {
            throw new Error(
                "Unable to discover AppScan tools from either provider.",
            );
        }

        const catalogue = buildToolCatalogue(local, remote);
        const selected = catalogue.find(
            (item) => item.capability === capability,
        );

        if (!selected) {
            throw new Error(
                `Unsupported or unavailable AppScan capability: ${capability}.`,
            );
        }

        if (selected.risk !== "write") {
            throw new Error(
                `Capability '${capability}' is not a write operation.`,
            );
        }

        const resolved = resolveToolProvider(selected, "local");

        const request: CapabilityRequest = {
            ...args,
            capability,
        };

        const normalizedArgs = normalizeCapabilityArguments(
            capability,
            resolved.provider,
            request,
        );

        const validate = this.ajv.compile(resolved.tool.inputSchema);

        if (!validate(normalizedArgs)) {
            const errors = (validate.errors ?? [])
                .map(
                    (error: ErrorObject) =>
                        `${error.instancePath || "/"}: ${error.message ?? "Invalid value"}`,
                )
                .join("; ");

            throw new Error(
                `Invalid arguments for ${resolved.tool.name}: ${errors}`,
            );
        }

        return {
            provider: resolved.provider,
            toolName: resolved.tool.name,
            normalizedArgs,
        };
    }

    /**
     * Execute a previously prepared write capability.
     * The provider and backend tool must match the approved preview.
     */
    async callApprovedCapability(
        capability: string,
        args: Record<string, unknown>,
        expectedProvider: AppScanProvider,
        expectedToolName: string,
    ): Promise<McpToolCallResult> {
        const preview = await this.prepareApprovedCapability(
            capability,
            args,
        );

        if (
            preview.provider !== expectedProvider ||
            preview.toolName !== expectedToolName
        ) {
            throw new Error(
                "The AppScan provider or backend tool has changed since approval. " +
                "The operation was not executed. Prepare it again and obtain approval.",
            );
        }

        return this.callProviderTool(
            expectedProvider,
            expectedToolName,
            preview.normalizedArgs,
        );
    }

    async callTool(
        name: string,
        args: Record < string, unknown >  = {}, ): Promise < McpToolCallResult > {
        return this.withLock(() => this.connector.callTool(name, args));
    }

    async switchProvider(
        provider: AppScanProvider, ): Promise < {
        provider: AppScanProvider;
        toolCount: number;
    }
     > {
        return this.withLock(async() => {
            if (provider === this.activeProvider) {
                return {
                    provider: this.activeProvider,
                    toolCount: (await this.connector.listTools()).length,
                };
            }

            const previousConnector = this.connector;
            let nextConnector: AppScanMcpConnector | undefined;

            try {
                nextConnector = this.connectorFactory(provider);
                await nextConnector.connect();

                const tools = await nextConnector.listTools();

                if (tools.length === 0) {
                    throw new Error(
`The ${provider} AppScan provider returned no tools.`, );
                }

                // Activate only after connecting and validating the new provider.
                this.connector = nextConnector;
                this.activeProvider = provider;

                try {
                    await previousConnector.disconnect();
                } catch (error) {
                    // The new provider is active; report cleanup failure without
                    // incorrectly rolling back a successful switch.
                    console.error(
                        "The previous AppScan connector could not be cleanly disconnected.",
                        error instanceof Error
                         ? error.message
                         : "Unknown error", );
                }

                return {
                    provider,
                    toolCount: tools.length,
                };
            } catch (error) {
                if (nextConnector && nextConnector !== this.connector) {
                    try {
                        await nextConnector.disconnect();
                    } catch {
                        // Preserve the original connection or validation error.
                    }
                }

                throw new Error(
`Could not switch to the ${provider} AppScan provider: ${
                    error instanceof Error
                     ? error.message
                     : "Unknown error"
}`, );
            }
        });
    }

    async findToolByNames(
        names: string[], ): Promise < McpTool | undefined > {
        return this.withLock(async() => {
            const tools = await this.connector.listTools();

            const normalizedNames = new Set(
                    names.map((name) => name.toLowerCase()), );

            return tools.find((tool) =>
                normalizedNames.has(tool.name.toLowerCase()), );
        });
    }

    async hasAvailableTool(
        names: string[],
        expectedProvider ?  : AppScanProvider, ): Promise < boolean > {
        return this.withLock(async() => {
            if (
                expectedProvider !== undefined &&
                this.activeProvider !== expectedProvider) {
                throw new Error(
                    `Provider mismatch: expected ${expectedProvider}, ` + 
`but active provider is ${this.activeProvider}.`, );
            }

            const tools = await this.connector.listTools();

            const normalizedNames = new Set(
                    names.map((name) => name.toLowerCase()), );

            return tools.some((tool) =>
                normalizedNames.has(tool.name.toLowerCase()), );
        });
    }

    async callFirstAvailableTool(
        names: string[],
        args: Record < string, unknown >  = {},
        expectedProvider ?  : AppScanProvider, ): Promise < McpToolCallResult > {
        return this.withLock(async() => {
            if (
                expectedProvider !== undefined &&
                this.activeProvider !== expectedProvider) {
                throw new Error(
`Provider mismatch: this operation was prepared for ${expectedProvider}, but the active provider is ${this.activeProvider}. The operation was not executed.`, );
            }

            // Discovery and execution use the same connector while holding the lock.
            const tools = await this.connector.listTools();

            const normalizedNames = new Set(
                    names.map((name) => name.toLowerCase()), );

            const tool = tools.find((item) =>
                    normalizedNames.has(item.name.toLowerCase()), );

            if (!tool) {
                throw new Error(
`The active ${this.activeProvider} AppScan provider does not support this operation. Expected one of: ${names.join(", ")}.`, );
            }

            return this.connector.callTool(tool.name, args);
        });
    }

    async callFirstAvailableToolIfSupported(
        names: string[],
        args: Record < string, unknown >  = {},
        expectedProvider ?  : AppScanProvider, ): Promise < McpToolCallResult > {
        return this.withLock(async() => {
            if (
                expectedProvider !== undefined &&
                this.activeProvider !== expectedProvider) {
                throw new Error(
                    `Provider mismatch: this operation was prepared for ${expectedProvider}, ` + 
`but the active provider is ${this.activeProvider}. The operation was not executed.`, );
            }

            const tools = await this.connector.listTools();

            const normalizedNames = new Set(
                    names.map((name) => name.toLowerCase()), );

            const tool = tools.find((item) =>
                    normalizedNames.has(item.name.toLowerCase()), );

            if (!tool) {
                throw new Error(
                    `The active ${this.activeProvider} AppScan provider does not support this operation. ` + 
`Expected one of: ${names.join(", ")}.`, );
            }

            return this.connector.callTool(tool.name, args);
        });
    }

    async disconnect(): Promise < void > {
        await this.withLock(() => this.connector.disconnect());
    }

    async findApplicationDiscoveryTool(): Promise < McpTool | undefined > {
        return this.withLock(async() => {
            const tools = await this.connector.listTools();

            return tools.find((tool) =>
                /^(get[_-]apps|list[_-]applications)$/i.test(tool.name), );
        });
    }

    async discoverApplications(
        args: Record < string, unknown >  = {}, ): Promise < McpToolCallResult > {
        return this.withLock(async() => {
            const tools = await this.connector.listTools();

            const tool = tools.find((item) =>
                    /^(get[_-]apps|list[_-]applications)$/i.test(item.name), );

            if (!tool) {
                throw new Error(
                    "The configured AppScan MCP server does not expose a recognized application discovery tool.", );
            }

            return this.connector.callTool(tool.name, args);
        });
    }
}
