import { randomUUID } from "node:crypto";
import type { AppScanService } from "./appscan-service.js";

export interface PreparedScan {
    appId: string;
    targetDir: string;
    scanType: "SAST" | "SCA";
    scanName ?  : string;
    comment ?  : string;
}

export interface ScanWorkflowResult {
    token ?  : string;
    text: string;
    isError ?  : boolean;
}

export class ScanWorkflow {
    private readonly pending = new Map < string,
    PreparedScan > ();

    constructor(private readonly service: AppScanService) {}

    prepare(input: PreparedScan): ScanWorkflowResult {
        const token = randomUUID();
        this.pending.set(token, {
            ...input
        });

        return {
            token,
            text: [
                "Scan prepared; no scan has been started.",
                `Confirmation token: ${token}`,
                `Application ID: ${input.appId}`,
                `Target directory: ${input.targetDir}`,
                `Scan type: ${input.scanType}`,
`Scan name: ${input.scanName ?? "(generated by AppScan)"}`,
                "",
                "Before execution, explain that source artifacts may be packaged and uploaded to AppScan, and that the underlying tool may generate configuration files or run a build.",
                "Obtain explicit user confirmation for these exact parameters before execution.",
            ].join("\n"),
        };
    }

    async execute(token: string): Promise < ScanWorkflowResult > {
        const prepared = this.pending.get(token);

        if (!prepared) {
            return {
                text: "No pending scan matches this token. Prepare the request again and obtain user confirmation.",
                isError: true,
            };
        }

        // Consume before invoking AppScan, preventing token reuse after failure.
        this.pending.delete(token);

        const result = await this.service.callTool("create-scan", {
            ...prepared,
            approved: true,
        });

        const text = result.content
            .map((item) => item.text ?? "")
            .filter(Boolean)
            .join("\n");

        return {
            text: text || "The AppScan tool returned no text content.",
            ...(result.isError ? {
                isError: true
            }
                 : {}),
        };
    }
}
