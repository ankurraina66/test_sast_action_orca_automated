    server.registerTool(

        description:
        "Prepare an AppScan scan request and return its exact parameters for user review. This tool must not start a scan. Obtain explicit user confirmation in the conversation before calling execute-prepared-scan.",
        inputSchema: {
            appId: z.string().uuid(),
            targetDir: z.string().min(1),
            scanType: z.enum(["SAST", "SCA"]).default("SAST"),
            scanName: z.string().min(1).optional(),
            comment: z.string().optional(),
        },
    },
        async({
            appId,
            targetDir,
            scanType,
            scanName,
            comment
        }) => {
        const token = crypto.randomUUID();
        const provider = service.getProvider();

        pendingScans.set(token, {
            appId,
            targetDir,
            scanType,

                scanName

            ...(comment !== undefined ? {
                comment
            }
                 : {}),
            provider,
        });

                "Scan prepared; no scan has been started.",
                `Confirmation token: ${token}`,
                `Application ID: ${appId}`,
                `Target directory: ${targetDir}`,
                `Scan type: ${scanType}`,
                `Scan name: ${scanName ?? "(generated by AppScan)"}`,
`Provider: ${provider}`,
