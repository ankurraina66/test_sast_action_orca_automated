const { RemoteHttpConnector } = require("./dist/mcp/remote-http-connector.js");

(async () => {
  const connector = new RemoteHttpConnector({
    endpoint: process.env.APP_SCAN_REMOTE_ENDPOINT,
    apiKey: process.env.APP_SCAN_API_KEY,
    requestTimeoutMs: 30000
  });

  try {
    await connector.connect();

    const result = await connector.callTool("get_apps", {
      top: 3,
      select: "Id,Name",
      count: true
    });

    console.log("get_apps isError:", result.isError);
    for (const item of result.content ?? []) {
      if (item.type === "text") {
        try {
          const data = JSON.parse(item.text);
          if (Array.isArray(data)) {
            console.log("Applications returned:", data.length);
            console.log(JSON.stringify(data.map(app => ({
              Id: app.Id ?? app.id,
              Name: app.Name ?? app.name
            })), null, 2));
          } else {
            console.log(JSON.stringify(data, null, 2));
          }
        } catch {
          console.log(item.text.slice(0, 3000));
        }
      }
    }
  } catch (error) {
    console.error("Read-only request failed:", error.message);
    process.exitCode = 1;
  } finally {
    try { await connector.disconnect(); } catch {}
  }
})();
