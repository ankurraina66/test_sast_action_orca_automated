from pathlib import Path

from pptx import Presentation
from pptx.util import Inches, Pt
from pptx.enum.text import PP_ALIGN
from pptx.enum.chart import XL_CHART_TYPE
from pptx.chart.data import CategoryChartData
from pptx.enum.shapes import MSO_SHAPE


class PresentationBuilder:
    def __init__(self):
        self.prs = Presentation()
        self.prs.slide_width = Inches(13.333)
        self.prs.slide_height = Inches(7.5)

    def _add_title(self, slide, title):
        box = slide.shapes.add_textbox(
            Inches(0.5), Inches(0.25), Inches(12.3), Inches(0.6)
        )
        p = box.text_frame.paragraphs[0]
        run = p.add_run()
        run.text = title
        run.font.size = Pt(28)
        run.font.bold = True

    def _add_text(self, slide, text, x, y, width, height,
                  font_size=18, bold=False):
        box = slide.shapes.add_textbox(
            Inches(x), Inches(y), Inches(width), Inches(height)
        )
        frame = box.text_frame
        frame.word_wrap = True
        p = frame.paragraphs[0]
        run = p.add_run()
        run.text = text
        run.font.size = Pt(font_size)
        run.font.bold = bold
        return box

    def add_summary_slide(self, summary, insights):
        slide = self.prs.slides.add_slide(self.prs.slide_layouts[6])
        self._add_title(slide, "Sales Performance — Executive Summary")

        row = summary.iloc[0]
        metrics = [
            ("Total Sales", row["Total Sales"], "number"),
            ("Total Profit", row["Total Profit"], "number"),
            ("Profit Margin", row["Profit Margin"], "percent"),
            ("Units Sold", row["Total Units"], "number"),
        ]
        x_positions = [0.5, 3.65, 6.8, 9.95]

        for index, (name, value, kind) in enumerate(metrics):
            x = x_positions[index]
            shape = slide.shapes.add_shape(
                MSO_SHAPE.ROUNDED_RECTANGLE,
                Inches(x), Inches(1.3), Inches(2.8), Inches(1.4)
            )
            shape.text_frame.clear()
            p = shape.text_frame.paragraphs[0]
            p.alignment = PP_ALIGN.CENTER
            run = p.add_run()
            run.text = f"{value:.1%}" if kind == "percent" else f"{value:,.0f}"
            run.font.size = Pt(24)
            run.font.bold = True

            p2 = shape.text_frame.add_paragraph()
            p2.alignment = PP_ALIGN.CENTER
            r2 = p2.add_run()
            r2.text = name
            r2.font.size = Pt(13)

        self._add_text(
            slide, "Key Insights", 0.6, 3.2, 5, 0.5, 22, True
        )

        y = 3.8
        for insight in insights:
            self._add_text(
                slide, "• " + insight, 0.8, y, 11.5, 0.5, 17
            )
            y += 0.55

    def add_region_slide(self, region_data):
        slide = self.prs.slides.add_slide(self.prs.slide_layouts[6])
        self._add_title(slide, "Sales Performance by Region")

        chart_data = CategoryChartData()
        chart_data.categories = region_data["Sales[Region]"].astype(str).tolist()
        chart_data.add_series(
            "Sales",
            [float(v) for v in region_data["Total Sales"].tolist()]
        )

        chart = slide.shapes.add_chart(
            XL_CHART_TYPE.BAR_CLUSTERED,
            Inches(1), Inches(1.5), Inches(11), Inches(5), chart_data
        ).chart

        chart.has_legend = False
        chart.value_axis.has_major_gridlines = True

    def add_product_slide(self, product_data):
        slide = self.prs.slides.add_slide(self.prs.slide_layouts[6])
        self._add_title(slide, "Sales Performance by Product")

        chart_data = CategoryChartData()
        chart_data.categories = product_data["Sales[Product]"].astype(str).tolist()
        chart_data.add_series(
            "Sales",
            [float(v) for v in product_data["Total Sales"].tolist()]
        )

        chart = slide.shapes.add_chart(
            XL_CHART_TYPE.COLUMN_CLUSTERED,
            Inches(1), Inches(1.5), Inches(11), Inches(5), chart_data
        ).chart

        chart.has_legend = False
        chart.value_axis.has_major_gridlines = True

    def add_product_table_slide(self, product_data):
        slide = self.prs.slides.add_slide(self.prs.slide_layouts[6])
        self._add_title(slide, "Product Performance Detail")

        rows = len(product_data) + 1
        cols = 4

        table_shape = slide.shapes.add_table(
            rows, cols,
            Inches(0.8), Inches(1.5), Inches(11.7), Inches(4.8)
        )
        table = table_shape.table

        headers = ["Product", "Sales", "Profit", "Units"]
        for col, header in enumerate(headers):
            table.cell(0, col).text = header

        for row_index, (_, row) in enumerate(product_data.iterrows(), start=1):
            table.cell(row_index, 0).text = str(row["Sales[Product]"])
            table.cell(row_index, 1).text = f'{row["Total Sales"]:,.0f}'
            table.cell(row_index, 2).text = f'{row["Total Profit"]:,.0f}'
            table.cell(row_index, 3).text = f'{row["Total Units"]:,.0f}'

    def add_insights_slide(self, insights):
        slide = self.prs.slides.add_slide(self.prs.slide_layouts[6])
        self._add_title(slide, "Key Takeaways")

        y = 1.4
        for index, insight in enumerate(insights):
            self._add_text(
                slide,
                f"{index + 1}. {insight}",
                0.8, y, 11.5, 0.7, 20
            )
            y += 0.85

    def save(self, path):
        Path(path).parent.mkdir(parents=True, exist_ok=True)
        self.prs.save(path)
