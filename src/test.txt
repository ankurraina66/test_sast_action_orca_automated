import { Client } from "@modelcontextprotocol/sdk/client/index.js";
import { StreamableHTTPClientTransport } from "@modelcontextprotocol/sdk/client/streamableHttp.js";
import type { Tool } from "@modelcontextprotocol/sdk/types.js";

import type {
AppScanMcpConnector,
McpTool,
McpToolCallResult,
} from "./connector.js";

export interface RemoteHttpConnectorOptions {
endpoint: string;
apiKey?: string;
requestTimeoutMs?: number;
}

export class RemoteHttpConnector implements AppScanMcpConnector {
private readonly client: Client;
private readonly transport: StreamableHTTPClientTransport;
private connected = false;

constructor(options: RemoteHttpConnectorOptions) {
this.client = new Client({
name: "hcl-appscan-connector-for-claude",
version: "0.1.0",
});

const headers: Record<string, string> = {};

if (options.apiKey) {
  headers["X-API-KEY"] = options.apiKey;
}

this.transport = new StreamableHTTPClientTransport(
  new URL(options.endpoint),
  {
    requestInit: {
      headers,
      signal: options.requestTimeoutMs
        ? AbortSignal.timeout(options.requestTimeoutMs)
        : undefined,
    },
  },
);

}

async connect(): Promise<void> {
if (this.connected) {
return;
}

await this.client.connect(this.transport);
this.connected = true;

}

async listTools(): Promise<McpTool[]> {
this.ensureConnected();

const result = await this.client.listTools();

return result.tools.map((tool: Tool) => ({
  name: tool.name,
  description: tool.description,
  inputSchema: tool.inputSchema as Record<string, unknown>,
}));

}

async callTool(
name: string,
args: Record<string, unknown>,
): Promise<McpToolCallResult> {
this.ensureConnected();

const result: unknown = await this.client.callTool({
  name,
  arguments: args,
});

if (
  typeof result !== "object" ||
  result === null ||
  !("content" in result) ||
  !Array.isArray(result.content)
) {
  throw new Error(
    `MCP tool "${name}" returned an unsupported result format.`,
  );
}

const content: McpToolCallResult["content"] = result.content.map(
  (item: unknown) => {
    if (typeof item !== "object" || item === null || !("type" in item)) {
      throw new Error(
        `MCP tool "${name}" returned an invalid content item.`,
      );
    }

    if (typeof item.type !== "string") {
      throw new Error(
        `MCP tool "${name}" returned a content item with an invalid type.`,
      );
    }

    if ("text" in item && typeof item.text === "string") {
      return { type: item.type, text: item.text };
    }

    return { type: item.type };
  },
);

return {
  content,
  isError:
    "isError" in result && typeof result.isError === "boolean"
      ? result.isError
      : undefined,
};

}

async disconnect(): Promise<void> {
if (!this.connected) {
return;
}

try {
  await this.client.close();
} finally {
  this.connected = false;
}

}

private ensureConnected(): void {
if (!this.connected) {
throw new Error("Remote AppScan MCP connector is not connected.");
}
}
}
