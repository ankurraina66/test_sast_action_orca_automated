import type {
  AppScanMcpConnector,
  McpTool,
  McpToolCallResult,
} from "../mcp/connector.js";

export class AppScanService {
  constructor(
    private readonly connector: AppScanMcpConnector,
  ) {}

  async connect(): Promise<void> {
    await this.connector.connect();
  }

  async listTools(): Promise<McpTool[]> {
    return this.connector.listTools();
  }

  async callTool(
    name: string,
    args: Record<string, unknown> = {},
  ): Promise<McpToolCallResult> {
    return this.connector.callTool(name, args);
  }

  async disconnect(): Promise<void> {
    await this.connector.disconnect();
  }

  async findApplicationDiscoveryTool(): Promise<McpTool | undefined> {
    const tools = await this.listTools();

    return tools.find((tool) =>
      /^(get[_-]apps|list[_-]applications)$/i.test(tool.name),
    );
  }

  async discoverApplications(
    args: Record<string, unknown> = {},
  ): Promise<McpToolCallResult> {
    const tool = await this.findApplicationDiscoveryTool();

    if (!tool) {
      throw new Error(
        "The configured AppScan MCP server does not expose a recognized application discovery tool.",
      );
    }

    return this.callTool(tool.name, args);
  }
}
