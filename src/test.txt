import { McpServer } from "@modelcontextprotocol/sdk/server/mcp.js";
import { StdioServerTransport } from "@modelcontextprotocol/sdk/server/stdio.js";
import { z } from "zod";
import { createAppScanService } from "./services/appscan-service-factory.js";
import { loadScanLifecycleConfig } from "./config/config.js";
import { DEFAULT_LIFECYCLE_OPTIONS, createRedactor, formatScanLifecycleResult, runScanLifecycle,  } from "./services/scan-lifecycle.js";
import type { ScanLifecycleResult, ScanOutcome,  } from "./services/scan-lifecycle.js";
import crypto from "node:crypto";

type AppScanProvider = "local" | "remote";

type PreparedScan = {
    appId: string;
    targetDir: string;
    scanType: "SAST" | "SCA";
    scanName ?  : string;
    comment ?  : string;
    provider: AppScanProvider;
};

type ScanExecutionState =
     | {
    status: "running";
    startedAt: string;
    phase ?  : string;
    scanId ?  : string;
    lastKnownStatus ?  : string;
}
 | {
    status: ScanOutcome;
    startedAt: string;
    completedAt: string;
    result ?  : ScanLifecycleResult;
    error ?  : string;
};

async function main(): Promise < void > {
    const service = createAppScanService();
    await service.connect();

    const server = new McpServer({
        name: "ai-coding-assistants",
        version: "0.1.0",
    });

    function textResult(result: {
        content: Array < {
            type: string;
            text ?  : string;
        }
         > ;
        isError ?  : boolean;
    }) {
        const text = result.content
            .map((item) => item.text ?? "")
            .filter(Boolean)
            .join("\n");

        return {
            content: [{
                    type: "text" as const,
                    text: text || "The AppScan tool returned no text content.",
                }
            ],
            ...(result.isError ? {
                isError: true
            }
                 : {}),
        };
    }

    function messageResult(text: string, isError = false) {
        return {
            content: [{
                    type: "text" as const,
                    text,
                }
            ],
            ...(isError ? {
                isError: true
            }
                 : {}),
        };
    }

    server.registerTool(
        "get-provider-status", {
        description: "Show which AppScan provider is currently active.",
        inputSchema: {},
    },
        async() => ({
            content: [{
                    type: "text" as const,
                    text: `Active AppScan provider: ${service.getProvider()}`,
                }
            ],
        }), );

    server.registerTool(
        "switch-provider", {
        description:
        "Switch the active AppScan backend between local STDIO and remote HTTP without restarting the Claude-facing MCP server.",
        inputSchema: {
            provider: z.enum(["local", "remote"]),
        },
    },
        async({
            provider
        }) => {
        try {
            const result = await service.switchProvider(provider);

            return messageResult([
                    `Active AppScan provider: ${result.provider}`, 
`Available backend tools: ${result.toolCount}`,
                    "Provider connection and tool discovery succeeded.",
                ].join("\n"));
        } catch (error) {
            return messageResult(
                error instanceof Error
                 ? error.message
                 : "Provider switching failed.",
                true, );
        }
    }, );

    server.registerTool(
        "create-app", {
        description:
        "Create an AppScan application using the Local STDIO backend only. First call without confirmToken to obtain the creation preview. Show the preview and obtain explicit user approval. Only then call again with the same application details and the returned confirmToken. Never create an application without explicit approval. This operation is not supported by the Remote HTTP backend.",
        inputSchema: {
            name: z.string().min(1),
            description: z.string().optional(),
            assetGroupId: z.string().optional(),
            confirmToken: z.string().optional(),
        },
    },
        async({
            name,
            description,
            assetGroupId,
            confirmToken
        }) => {
        // Application creation is supported only by Local MCP.
        if (service.getProvider() !== "local") {
            return messageResult(
                "Application creation is supported only by the Local AppScan MCP backend. Switch to the local provider and try again.",
                true, );
        }

        try {
            const result = await service.callFirstAvailableTool(
                    ["create-app", "create_app"], {
                    name,
                    ...(description !== undefined ? {
                        description
                    }
                         : {}),
                    ...(assetGroupId !== undefined ? {
                        assetGroupId
                    }
                         : {}),
                    ...(confirmToken !== undefined ? {
                        confirmToken
                    }
                         : {}),
                },
                    "local", );

            return textResult(result);
        } catch (error) {
            return messageResult(
                error instanceof Error
                 ? error.message
                 : "AppScan application creation failed.",
                true, );
        }
    }, );

    server.registerTool(
        "get-apps", {
        description: "Retrieve AppScan applications.",
        inputSchema: {
            filter: z.string().optional(),
            top: z.number().optional(),
        },
    },
        async(args) => {
        console.error("[AppScan diagnostic] get-apps invoked");

        try {
            const result = await service.discoverApplications({
                ...(args.filter !== undefined
                     ? {
                    filter: args.filter
                }
                     : {}),
                ...(args.top !== undefined
                     ? {
                    top: args.top
                }
                     : {}),
            });

            console.error(
`[AppScan diagnostic] get-apps completed; isError=${result.isError ?? false}`, );

            return textResult(result);
        } catch (error) {
            console.error(
                "[AppScan diagnostic] get-apps failed:",
                error instanceof Error ? error.message : "Unknown error", );
            throw error;
        }
    }, );

    server.registerTool(
        "get-scans", {
        description:
        "Retrieve AppScan scans. Optionally filter by applicationId (GUID) and an additional OData filter.",
        inputSchema: {
            applicationId: z.string().uuid().optional(),
            filter: z.string().optional(),
            top: z.number().int().positive().optional(),
        },
    },
        async({
            applicationId,
            filter,
            top
        }) => {
        const filters: string[] = [];

        if (applicationId) {
            filters.push(`AppId eq ${applicationId}`);
        }

        if (filter?.trim()) {
            filters.push(`(${filter.trim()})`);
        }

        const result = await service.callFirstAvailableTool(
                ["get-scans", "get_scans"], {
                ...(filters.length
                     ? {
                    filter: filters.join(" and ")
                }
                     : {}),
                ...(top !== undefined ? {
                    top
                }
                     : {}),
            }, );

        return textResult(result);
    }, );

    server.registerTool(
        "get-scan-issues", {
        description: "Retrieve security findings for a specific AppScan scan.",
        inputSchema: {
            scanId: z.string().min(1),
            filter: z.string().optional(),
            top: z.number().int().positive().optional(),
        },
    },
        async({
            scanId,
            filter,
            top
        }) => {
        const provider = service.getProvider();

        if (provider === "local") {
            const result = await service.callFirstAvailableTool(
                    ["get-scan-issues", "get_scan_issues"], {
                    scanId,
                    ...(filter !== undefined ? {
                        filter
                    }
                         : {}),
                    ...(top !== undefined ? {
                        top
                    }
                         : {}),
                },
                    "local", );

            return textResult(result);
        }

        const result = await service.callFirstAvailableTool(
                ["get_issues"], {
                scope: "Scan",
                scopeId: scanId,
                ...(filter !== undefined ? {
                    filter
                }
                     : {}),
                ...(top !== undefined ? {
                    top
                }
                     : {}),
            },
                "remote", );

        return textResult(result);
    }, );

    server.registerTool(
        "get-issue-details", {
        description: "Retrieve detailed information about an AppScan finding.",
        inputSchema: {
            issueId: z.string().min(1),
        },
    },
        async({
            issueId
        }) => {
        const result = await service.callFirstAvailableTool(
                ["get-issue-details", "get_issue_details"], {
                issueId
            }, );

        return textResult(result);
    }, );

    const pendingScans = new Map < string,
    PreparedScan > ();
    const executingScanTokens = new Set < string > ();
    const scanExecutions = new Map < string,
    ScanExecutionState > ();
    const lifecycleConfig = loadScanLifecycleConfig();

    server.registerTool(
        "prepare-scan", {
        description:
        "Prepare an AppScan scan request and return its exact parameters for user review. This tool must not start a scan. Obtain explicit user confirmation in the conversation before calling execute-prepared-scan.",
        inputSchema: {
            appId: z.string().uuid(),
            targetDir: z.string().min(1),
            scanType: z.enum(["SAST", "SCA"]).default("SAST"),
            scanName: z.string().min(1).optional(),
            comment: z.string().optional(),
        },
    },
        async({
            appId,
            targetDir,
            scanType,
            scanName,
            comment
        }) => {
        const token = crypto.randomUUID();
        const provider = service.getProvider();

        pendingScans.set(token, {
            appId,
            targetDir,
            scanType,
            ...(scanName !== undefined ? {
                scanName
            }
                 : {}),
            ...(comment !== undefined ? {
                comment
            }
                 : {}),
            provider,
        });

        return messageResult([
                "Scan prepared; no scan has been started.",
                `Confirmation token: ${token}`, 
                `Application ID: ${appId}`, 
                `Target directory: ${targetDir}`, 
                `Scan type: ${scanType}`, 
                `Scan name: ${scanName ?? "(generated by AppScan)"}`, 
`Provider: ${provider}`,
                "",
                "Before execution, explain that source artifacts may be packaged and uploaded to AppScan, and that the underlying tool may generate configuration files or run a build.",
                "Obtain explicit user confirmation for these exact parameters before invoking execute-prepared-scan.",
            ].join("\n"));
    }, );

    server.registerTool(
        "execute-prepared-scan", {
        description:
        "Execute a previously prepared AppScan scan only after explicit user confirmation of its exact parameters. Returns promptly; use get-scan-execution-status to retrieve its eventual result. Never call this tool merely because a preparation token exists.",
        inputSchema: {
            confirmationToken: z.string().uuid(),
            // Default true: wait for a confirmed outcome and return the
            // consolidated result. False restores the earlier detached mode.
            waitForCompletion: z.boolean().default(true),
        },
    },
        async({
            confirmationToken,
            waitForCompletion
        }) => {
        const prepared = pendingScans.get(confirmationToken);

        if (!prepared) {
            return messageResult(
                "No pending scan matches this token. Prepare the request again and obtain user confirmation.",
                true, );
        }

        if (executingScanTokens.has(confirmationToken)) {
            return messageResult(
                "This confirmation token is already being processed. Do not submit it again.",
                true, );
        }

        if (prepared.provider !== service.getProvider()) {
            return messageResult(
`This scan was prepared for the ${prepared.provider} provider, but the active provider is ${service.getProvider()}. The scan was not started. Prepare the request again and obtain explicit confirmation.`,
                true, );
        }

        // Claim and consume the approval token before any work starts, so a
        // token can never start a second scan.
        executingScanTokens.add(confirmationToken);
        pendingScans.delete(confirmationToken);

        const startedAt = new Date().toISOString();

        scanExecutions.set(confirmationToken, {
            status: "running",
            startedAt,
            phase: "creating",
        });

        const redact = createRedactor();

        // One lifecycle per token: create once, poll until a confirmed outcome,
        // then collect results. It is never retried or re-submitted.
        const lifecycle = (async() => {
            try {
                const result = await runScanLifecycle(
                        prepared, {
                        callTool: (names, args) =>
                        service.callFirstAvailableToolIfSupported(
                            names,
                            args,
                            prepared.provider, ),
                        redact,
                        onProgress: (progress) => {
                            scanExecutions.set(confirmationToken, {
                                status: "running",
                                startedAt,
                                phase: progress.phase,
                                ...(progress.scanId !== undefined
                                     ? {
                                    scanId: progress.scanId
                                }
                                     : {}),
                                ...(progress.lastKnownStatus !== undefined
                                     ? {
                                    lastKnownStatus: progress.lastKnownStatus
                                }
                                     : {}),
                            });
                        },
                    }, {
                        ...lifecycleConfig,
                        ...DEFAULT_LIFECYCLE_OPTIONS,
                    }, );

                scanExecutions.set(confirmationToken, {
                    status: result.outcome,
                    startedAt,
                    completedAt: result.endedAt,
                    result,
                });

                return result;
            } finally {
                executingScanTokens.delete(confirmationToken);
            }
        })().catch((error: unknown) => {
            const message = redact(
                    error instanceof Error ? error.message : "Unknown error", );

            console.error(
`[AppScan scan execution] token=${confirmationToken} ended unexpectedly: ${message}`, );

            scanExecutions.set(confirmationToken, {
                status: "unconfirmed",
                startedAt,
                completedAt: new Date().toISOString(),
                error: message,
            });

            return undefined;
        });

        if (!waitForCompletion) {
            return messageResult([
                    "Scan execution was accepted by the connector and is running in the background.",
`Confirmation token: ${confirmationToken}`,
                    "Use get-scan-execution-status with this token to retrieve the final result.",
                    "Do not submit another scan while the status is running or uncertain.",
                ].join("\n"));
        }

        const outcome = await lifecycle;

        if (!outcome) {
            return messageResult([
                    "Outcome: UNCONFIRMED",
                    "The scan lifecycle ended unexpectedly. Do not submit another scan.",
`Confirmation token: ${confirmationToken}`,
                    "Use get-scan-execution-status with this token to see what is known.",
                ].join("\n"), true);
        }

        return messageResult(
            formatScanLifecycleResult(outcome),
            outcome.outcome !== "completed", );
    }, );

    server.registerTool(
        "get-scan-execution-status", {
        description:
        "Retrieve the status and final result of a previously approved scan execution. This tool never starts a scan.",
        inputSchema: {
            confirmationToken: z.string().uuid(),
        },
    },
        async({
            confirmationToken
        }) => {
        const execution = scanExecutions.get(confirmationToken);

        if (!execution) {
            return messageResult(
                "No execution record exists for this token. Do not retry the scan automatically; verify the AppScan backend first.",
                true, );
        }

        if (execution.status === "running") {
            return messageResult(JSON.stringify({
                    status: execution.status,
                    startedAt: execution.startedAt,
                    ...(execution.phase !== undefined
                         ? {
                        phase: execution.phase
                    }
                         : {}),
                    ...(execution.scanId !== undefined
                         ? {
                        scanId: execution.scanId
                    }
                         : {}),
                    ...(execution.lastKnownStatus !== undefined
                         ? {
                        lastKnownStatus: execution.lastKnownStatus
                    }
                         : {}),
                    confirmationToken,
                }, null, 2));
        }

        const finished = execution.status === "completed";

        return messageResult(JSON.stringify({
                status: execution.status,
                startedAt: execution.startedAt,
                completedAt: execution.completedAt,
                confirmationToken,
                ...(execution.result !== undefined
                     ? {
                    result: execution.result
                }
                     : {}),
                ...(execution.error !== undefined
                     ? {
                    error: execution.error
                }
                     : {}),
                ...(finished
                     ? {}
                     : {
                    warning:
                    "The final backend outcome may require verification. Do not automatically start another scan.",
                }),
            }, null, 2), !finished);
    }, );

    const transport = new StdioServerTransport();

    try {
        await server.connect(transport);
        console.error("HCL AppScan Connector for Claude is running.");
    } catch (error) {
        await service.disconnect();
        throw error;
    }

    const shutdown = async(): Promise < void >  => {
        try {
            await server.close();
        } finally {
            await service.disconnect();
        }
    };

    process.once("SIGINT", () => {
        void shutdown().catch((error: unknown) => {
            console.error("Shutdown failed:", error);
            process.exitCode = 1;
        });
    });

    process.once("SIGTERM", () => {
        void shutdown().catch((error: unknown) => {
            console.error("Shutdown failed:", error);
            process.exitCode = 1;
        });
    });
}

main().catch((error: unknown) => {
    console.error(
        "Failed to start HCL AppScan Connector for Claude:",
        error instanceof Error ? error.message : "Unknown error", );
    process.exitCode = 1;
});
