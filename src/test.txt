    async callFirstAvailableTool(
        names: string[],
        args: Record < string, unknown >  = {},
        expectedProvider ?  : AppScanProvider, ): Promise < McpToolCallResult > {
        return this.withLock(async() => {
            if (
                expectedProvider !== undefined &&
                this.activeProvider !== expectedProvider) {
                throw new Error(
`Provider mismatch: this operation was prepared for ${expectedProvider}, but the active provider is ${this.activeProvider}. The operation was not executed.`, );
            }

            // Discovery and execution use the same connector while holding the lock.
            const tools = await this.connector.listTools();
            const normalizedNames = new Set(
                    names.map((name) => name.toLowerCase()), );
            const tool = tools.find((item) =>
                    normalizedNames.has(item.name.toLowerCase()), );

            if (!tool) {
                throw new Error(
`The active ${this.activeProvider} AppScan provider does not support this operation. Expected one of: ${names.join(", ")}.`, );
            }

            return this.connector.callTool(tool.name, args);
        });
    }

    async disconnect(): Promise < void > {
        await this.withLock(() => this.connector.disconnect());
    }
