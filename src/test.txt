it("passes the API key in the X-API-KEY header", async () => {
  new RemoteHttpConnector({
    endpoint: "https://cloud.appscan.com/mcp",
    apiKey: "test-key",
    requestTimeoutMs: 5000,
  });

  const transportOptions = mockTransport.mock.calls.at(-1)?.[1] as {
    fetch: typeof fetch;
  };

  const mockFetch = vi.fn().mockResolvedValue(new Response("OK"));
  const originalFetch = globalThis.fetch;

  vi.stubGlobal("fetch", mockFetch);

  try {
    await transportOptions.fetch("https://cloud.appscan.com/mcp", {
      method: "POST",
      headers: { "content-type": "application/json" },
    });

    const requestHeaders = new Headers(mockFetch.mock.calls[0][1]?.headers);
    expect(requestHeaders.get("X-API-KEY")).toBe("test-key");
  } finally {
    vi.stubGlobal("fetch", originalFetch);
  }
});
