import { McpServer } from "@modelcontextprotocol/sdk/server/mcp.js";
import { StdioServerTransport } from "@modelcontextprotocol/sdk/server/stdio.js";
import { configureReportsRoots, resolveReportsDir } from "./helpers/reports-dir.js";
import { log, validateRuntimeConfig } from "./runtime.js";
import { registerAllTools } from "./tools/registry.js";
// Create server instance
const server = new McpServer({
    name: "AppScan",
    title: "AppScan",
    description: "Connects AI assistants to HCL AppScan on Cloud to manage applications, scans, security issues, and reports.",
    version: "1.0.0",
});
registerAllTools(server);
configureReportsRoots(async () => {
    if (!server.server.getClientCapabilities()?.roots)
        return undefined;
    const response = await server.server.listRoots(undefined, { timeout: 3_000 });
    return response.roots[0]?.uri;
});
server.server.oninitialized = () => {
    void resolveReportsDir().catch((error) => {
        console.error(`Reports directory configuration error: ${error instanceof Error ? error.message : String(error)}`);
    });
};
/**
 * Prompt: Pre-scan setup instructions
 * This prompt provides guidance before starting a security scan
 */
server.prompt("pre-scan-setup", "Instructions to follow before creating an AppScan security scan", {}, async () => {
    return {
        messages: [
            {
                role: "user",
                content: {
                    type: "text",
                    text: `BEFORE STEP 1 - PRE-SCAN CHECKLIST:

Before running the create-scan tool, ensure the following:
First check if in the root directory of your target codebase there is an existing AppScan configuration file (appscan-config.xml). If such a file exists, review its contents to ensure it aligns with your scanning requirements. You may choose to use this existing configuration by providing its path to the create-scan tool via the configFile parameter.

For SAST, the tool uses an existing targetDir/appscan-config.xml or generates a mode-aware configuration. A missing explicitly supplied configFile is an error. Relative configuration targets resolve against targetDir.
Use scanMode=auto unless the user requests source-code-only or build-output. Mixed or unknown targets require a deliberate scope/mode choice; never scan just the frontend by accident.
Java/.NET build-output scans need prebuilt artifacts; C/C++ needs a supported Visual Studio solution. .NET/C++ build-output preparation requires Windows. SAST does not run build commands.
If a build is needed, an AI agent with terminal access may inspect and run the project's documented build command with separate user approval, verify success and artifacts, and then request scan approval. Stop on build failure; never silently switch to source-only. A client without terminal access needs developer/CI-prepared artifacts.
Configuration documentation: https://help.hcl-software.com/appscan/ASoC/src_irx_gen_config.html
Then proceed with create-scan.`,
                },
            },
        ],
    };
});
/**
 * Prompt: Create AppScan scan workflow
 * This prompt helps assistants map natural scan requests to the create-scan tool.
 */
server.prompt("create-appscan-scan", "Use this workflow when the user asks to create, run, start, or launch an HCL AppScan security scan", {}, async () => {
    return {
        messages: [
            {
                role: "user",
                content: {
                    type: "text",
                    text: `When the user asks to create a scan, run a scan, start a SAST scan, start an SCA scan, start a DAST scan, scan this project, security test this codebase, analyze this workspace for vulnerabilities, or run AppScan against the current project or a target URL, use the create-scan tool. Do this even when the user does not explicitly say "use the MCP server".

Workflow:
1. Use the current workspace folder as targetDir when the user says "this project" or does not provide a path.
2. The server does not retain selections between chat sessions. Reuse an application ID only when it was explicitly established in the current conversation, and pass that ID to create-scan.
3. If the user identified an application by name or another selector but did not provide its ID, use get-apps with a filter derived from that selector. Continue only if the result is complete and contains exactly one match. A top limit, unknown total, or unfiltered singleton is not enough. If the name has no match, show the other existing applications returned by get-apps; fetch an unfiltered list if needed.
4. If no app selector is available or selection remains unresolved, show actual application names and IDs alongside "Create a new application" and "Enter another application name or ID" in a selection UI or visible list, then wait for the user's choice. Do this even if the unfiltered list contains only one application. Never replace these choices with a create-only yes/no question. Do not infer an AppScan application selector from the project folder name. Creating an application requires separate explicit approval.
5. Never infer missing application details. Never invent an application name, asset group, ID, or approval.
6. Never create an application or a scan until the user has explicitly approved that specific write action in clear words.
7. If the user wants a new app for the scan, call create-app. If a preview token is returned, show the preview, get explicit user agreement, then call create-app again with that confirmToken.
8. If the supplied appId is missing or invalid, do not create a new app automatically. Present the existing applications and the create-new option as in step 4, rather than merely saying existing applications were found.
9. Treat create-app and create-scan as two separate writes. Approval for one does not imply approval for the other unless the user clearly grants both in the same turn.
10. Default scanType to SAST unless the user asks for SCA/dependency/software composition analysis, or asks for DAST/dynamic/web URL scanning.
11. Set personalScan=true when the user requests a personal scan; this option applies to SAST, SCA, and DAST. Otherwise leave it false.
12. Default detailedLogs to false. Only set detailedLogs to true when the user explicitly asks for verbose or troubleshooting logs; detailed logs can consume significantly more chat/model tokens.
13. Default conserveFiles to false so local generated scan artifacts are deleted after the scan is created or after the scan setup fails.
14. Set conserveFiles to true when the user asks to keep, conserve, retain, save, or inspect generated files, artifacts, zip, IRX, logs, debug output, or anything similar.
15. If the user wants the lowest-token scan flow, set wait=false and detailedLogs=false.
16. Call create-scan only after appId is known and scan approval is explicit.
17. Never bypass the approval flow. If preview-confirm is used, require the returned confirmToken from the immediately prior preview.
18. Never reuse old confirmToken values. If token validation fails or expires, request a fresh preview and token.
19. For DAST scans, do not poll for completion from create-scan. Return immediately after scan creation with the scan status URL/API details and explain that DAST can run for a while.
20. For any asynchronously returned scan, include its current execution percentage and status when available, include the scan overview link, and ask whether the user would like a later status check. Do not imply that a background follow-up is scheduled.
21. If the user asks for a later check, call get-scans filtered by the exact scan ID and report the latest execution status and percentage.
22. For SAST, leave scanMode=auto unless the user requests source-code-only or build-output. If mixed or unknown targets are reported, select narrower targets or obtain an explicit mode choice; do not omit components.
23. SAST never executes project builds. Java/.NET need prebuilt artifacts for build-output scans; C/C++ needs a supported Visual Studio solution. .NET/C++ build-output preparation requires Windows.
24. With terminal tools and separate build approval, inspect and run the project's documented build command, verify success and artifacts, then call create-scan. Without terminal access, ask for developer/CI-prepared artifacts. Never downgrade to source-only after failure unless the user explicitly chooses it.
25. buildBeforeScan is SCA-only. Keep SCA/DAST behavior independent of SAST scanMode. Explain source-only versus build-output trade-offs and report the selected SAST mode.`,
                },
            },
        ],
    };
});
/**
 * Main function to run the server
 */
async function main() {
    const config = validateRuntimeConfig();
    const transport = new StdioServerTransport();
    await server.connect(transport);
    log("info", "server.started", { apiBaseUrl: config.apiBaseUrl, credentialsConfigured: true });
    let shuttingDown = false;
    const shutdown = async (signal) => {
        if (shuttingDown)
            return;
        shuttingDown = true;
        log("info", "server.stopping", { signal });
        await server.close();
        process.exitCode = 0;
    };
    process.once("SIGINT", () => void shutdown("SIGINT"));
    process.once("SIGTERM", () => void shutdown("SIGTERM"));
}
main().catch((error) => {
    log("error", "server.start_failed", { error: error instanceof Error ? error.message : String(error) });
    process.exitCode = 1;
});
