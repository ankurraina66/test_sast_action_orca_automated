package com.hcl.appscan.intellij.plugin.ui;

import com.google.common.io.Resources;
import com.hcl.appscan.intellij.plugin.MessagesBundle;
import com.hcl.appscan.intellij.plugin.auth.asoc.LoginUtility;
import com.hcl.appscan.intellij.plugin.handler.ProjectSettingsState;
import com.hcl.appscan.intellij.plugin.handler.asoc.ApplicationHandler;
import com.hcl.appscan.intellij.plugin.listener.asoc.IMessageTopics;
import com.hcl.appscan.intellij.plugin.util.CodeSweepUtils;
import com.hcl.appscan.intellij.plugin.util.OmniaUtil;
import com.hcl.appscan.intellij.plugin.util.UIComponentsUtil;
import com.hcl.appscan.sdk.auth.IAuthenticationProvider;
import com.intellij.ide.BrowserUtil;
import com.intellij.openapi.application.ApplicationManager;
import com.intellij.openapi.diagnostic.Logger;
import com.intellij.openapi.project.Project;
import com.intellij.ui.JBColor;
import com.intellij.ui.components.JBScrollPane;
import com.intellij.ui.jcef.JBCefBrowser;
import com.intellij.ui.jcef.JBCefBrowserBase;
import com.intellij.ui.jcef.JBCefJSQuery;
import com.intellij.util.ui.UIUtil;
import org.cef.browser.CefBrowser;
import org.cef.handler.CefLoadHandler;
import org.cef.handler.CefLoadHandlerAdapter;

import javax.swing.*;
import java.awt.*;
import java.beans.PropertyChangeEvent;
import java.beans.PropertyChangeListener;
import java.io.File;
import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;

public class HowToFixPanel implements PropertyChangeListener {
	private Object tempObject;
	private Object tempIssueObject;
	private JBCefBrowser browser;
	private ProjectSettingsState settingsState;
	protected JPanel panel;
	protected ApplicationHandler applicationHandler;
	protected IAuthenticationProvider authenticationHandler;
	protected Project project;
	protected static String issueTypeId = null;
	private static Logger logger = Logger.getInstance(HowToFixPanel.class);
	private JBCefJSQuery myJSQueryOpenInBrowser;
	private static String ourJSCodeToInject;
	private CefLoadHandler myCefLoadHandler;

	public HowToFixPanel(final Project project) {
		this.project = project;
		this.authenticationHandler = LoginUtility.getASoCHandler();
		this.applicationHandler = ApplicationHandler.getASoCApplicationHandler(project);
		this.settingsState = ProjectSettingsState.getInstance(project);

		createUIComponents();
		refreshUIComponents(null, null);
		subscribeToListener();
		UIManager.addPropertyChangeListener(this);
	}

	static {
		try {
			ourJSCodeToInject = Resources.toString(HowToFixPanel.class.getResource("/js/howtofix_script.js"), StandardCharsets.UTF_8);
		}
		catch (IOException e) {
			logger.warn("Failed to load script.js.", e);
		}
	}

	public static String getIssueTypeId() {
		return issueTypeId;
	}

	private void subscribeToListener() {
		project.getMessageBus().connect().subscribe(IMessageTopics.APPSCAN_DETAILS_LISTENER_TOPIC, this::refreshUIComponents);
	}

	private void createUIComponents() {
		this.panel = UIComponentsUtil.getDefaultXAxisPanel();
		this.browser = new JBCefBrowser();

		browser.getComponent().setPreferredSize(new Dimension(10, 10));
		browser.setPageBackgroundColor(UIComponentsUtil.getHtmlHexadecimalColorCode(JBColor.background()));

		myJSQueryOpenInBrowser = JBCefJSQuery.create((JBCefBrowserBase) browser);
		browser.getJBCefClient().addLoadHandler(myCefLoadHandler = new CefLoadHandlerAdapter() {
			@Override
			public void onLoadingStateChange(CefBrowser cefBrowser, boolean isLoading, boolean canGoBack, boolean canGoForward) {
				if (ourJSCodeToInject != null) {
					cefBrowser.executeJavaScript(ourJSCodeToInject, browser.getCefBrowser().getURL(), 0);
					cefBrowser.executeJavaScript("window.__Appscan.openInBrowserCallback = link => {"
									+ myJSQueryOpenInBrowser.inject("link") + "}",
							browser.getCefBrowser().getURL(), 0);
				}
			}
		}, browser.getCefBrowser());
		myJSQueryOpenInBrowser.addHandler((link) -> {
			if (link != null)
				BrowserUtil.browse(link);
			return null;
		});

		JPanel appPanel = new JPanel();
		appPanel.setLayout(new BoxLayout(appPanel, BoxLayout.X_AXIS));
		JBScrollPane scrollPane = new JBScrollPane(browser.getComponent(), JScrollPane.VERTICAL_SCROLLBAR_AS_NEEDED, JScrollPane.HORIZONTAL_SCROLLBAR_AS_NEEDED);
		scrollPane.setViewportView(browser.getComponent());
		appPanel.add(scrollPane);
		panel.add(appPanel);
		panel.updateUI();
	}

	public JPanel getComponent() {
		return panel;
	}

	protected void refreshUIComponents(Object object, Object issueObject) {
		this.tempObject = object;
		this.tempIssueObject = issueObject;
		File libFolder = OmniaUtil.getLibFolder();
		String templatePath = (UIUtil.isUnderDarcula() ? UIConstants.DARK_TEMPLATE_FOLDER : UIConstants.TEMPLATE_FOLDER);
		File templateDir = new File(libFolder, templatePath);
		String omniaUrl = templateDir.getAbsolutePath() + (UIUtil.isUnderDarcula() ? UIConstants.HOW_TO_FIX_HTML_DARK : UIConstants.HOW_TO_FIX_HTML);
		if (object instanceof String && CodeSweepUtils.SHOW_CODESWEEP_RULES_INFO.equals(object)) {
			omniaUrl = templateDir.getAbsolutePath() + (UIUtil.isUnderDarcula() ? UIConstants.CODESWEEP_RULES_HTML_DARK : UIConstants.CODESWEEP_RULES_HTML);
			object = null;
		}
		String mode = UIUtil.isUnderDarcula() ? UIConstants.DARK_MODE : UIConstants.LIGHT_MODE;
		if(object != null) {
			HowToFixPanel.issueTypeId = OmniaUtil.extractIssueTypeId(object, project);
			omniaUrl = OmniaUtil.getOmniaUrl(object, omniaUrl, mode, issueTypeId);
			browser.loadURL(omniaUrl);
		} else {
			try {
				Path path = Paths.get(omniaUrl);
				String fileContent = Files.readString(path);
				browser.loadHTML(fileContent);
			} catch (Exception e) {
				logger.error(MessagesBundle.message("toolwindow.issuedetails.dataflow.templatefilenotfound"), e);
			}
		}
	}

	@Override
	public void propertyChange(PropertyChangeEvent propertyChangeEvent) {
		if (UIConstants.LOOKANDFEEL_EVENT_NAME.equals(propertyChangeEvent.getPropertyName())) {
			ApplicationManager.getApplication().invokeLater(() -> {
				browser.setPageBackgroundColor(UIComponentsUtil.getHtmlHexadecimalColorCode(JBColor.background()));
				refreshUIComponents(tempObject, tempIssueObject);
			});
		}
	}
