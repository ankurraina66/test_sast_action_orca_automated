import type { ConnectorConfig } from "../config/config.js";
import type { AppScanMcpConnector } from "./connector.js";
import { LocalStdioConnector } from "./local-stdio-connector.js";
import { RemoteHttpConnector } from "./remote-http-connector.js";

export function createConnector(
  config: ConnectorConfig,
): AppScanMcpConnector {
  if (config.provider === "local") {
    if (!config.local) {
      throw new Error(
        "Local connector configuration is missing.",
      );
    }

    return new LocalStdioConnector({
      command: config.local.command,
      args: config.local.args,
      requestTimeoutMs: config.requestTimeoutMs,
      env: {
        APPSCAN_API_BASE: config.local.apiBase,
        ...(config.local.keyId
          ? { APPSCAN_KEY_ID: config.local.keyId }
          : {}),
        ...(config.local.keySecret
          ? { APPSCAN_KEY_SECRET: config.local.keySecret }
          : {}),
      },
    });
  }

  if (!config.remote) {
    throw new Error(
      "Remote connector configuration is missing.",
    );
  }

  return new RemoteHttpConnector({
    endpoint: config.remote.endpoint,
    apiKey: config.remote.apiKey,
    requestTimeoutMs: config.requestTimeoutMs,
  });
}
