
$headers = @{
    "X-Api-Key" = "$($env:APPSCAN_KEY_ID):$($env:APPSCAN_KEY_SECRET)"
    "Content-Type" = "application/json"
}

$response = Invoke-RestMethod `
    -Uri "$($env:APPSCAN_API_BASE)/api/v4/Scans?`$top=1" `
    -Headers $headers `
    -Method Get

$scans = if ($response.Items) { @($response.Items) } else { @($response) }
$scan = $scans | Select-Object -First 1

if (-not $scan) {
    Write-Host "No scan returned."
} else {
    Write-Host "`nTop-level scan fields:"
    $scan.PSObject.Properties.Name

    Write-Host "`nApplication-related fields:"
    $scan.PSObject.Properties |
        Where-Object { $_.Name -match 'app|application|project' } |
        ForEach-Object {
            $value = $_.Value
            if ($null -eq $value) {
                "$($_.Name) = null"
            } elseif ($value -is [string] -or $value -is [ValueType]) {
                "$($_.Name) = $value"
            } else {
                "$($_.Name) = <$($value.GetType().Name)>"
                if ($value -is [System.Collections.IDictionary]) {
                    "  Nested fields: $($value.Keys -join ', ')"
                } else {
                    "  Nested fields: $($value.PSObject.Properties.Name -join ', ')"
                }
            }
        }

    Write-Host "`nMetadata candidates:"
    foreach ($field in @('ScanType', 'Type', 'Created', 'CreatedAt', 'LatestExecution')) {
        $property = $scan.PSObject.Properties[$field]
        if ($property) {
            if ($field -eq 'LatestExecution' -and $null -ne $property.Value) {
                "LatestExecution fields: $($property.Value.PSObject.Properties.Name -join ', ')"
            } else {
                "$field = $($property.Value)"
            }
        }
    }
}
