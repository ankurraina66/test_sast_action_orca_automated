function validateCreateScanInput(input) {
    if (input.scanType !== "SAST" && input.scanMode !== "auto") {
        throw new Error("scanMode is only supported for SAST scans.");
    }
    if (input.scanType === "SAST" && input.buildBeforeScan) {
        throw new Error("buildBeforeScan is not supported for SAST. Prepare required artifacts using a separately approved terminal/CI build, then retry. Do not fall back to source-only without explicit approval.");
    }
    if (input.scanType === "DAST") {
        if (!input.scanUrl) {
            throw new Error("scanUrl is required when scanType is DAST.");
        }
        if (input.networkType === "private" && !input.presenceId) {
            throw new Error("presenceId is required when scanType is DAST and networkType is private.");
        }
        if (input.loginMethod === "Required" && (!input.loginUser || !input.loginPassword)) {
            throw new Error("loginUser and loginPassword are required when scanType is DAST and loginMethod is Required.");
        }
        if (input.loginMethod === "Record" && !input.loginSequenceFileId) {
            throw new Error("loginSequenceFileId is required when scanType is DAST and loginMethod is Record.");
        }
        if (input.buildBeforeScan) {
            throw new Error("buildBeforeScan is not supported for DAST scans.");
        }
        return;
    }
    if (!input.targetDir) {
        throw new Error("targetDir is required when scanType is SAST or SCA.");
    }
}
function buildApprovalSummary(input, appName) {
    const effectiveConfig = input.configFile || "<generated>/appscan-config.xml";
    const effectiveWorkDir = input.workDir || "<system-temp>/appscan-mcp";
    if (input.scanType === "DAST") {
        return [
            `Tool: create-scan`,
            `Target environment: AppScan on Cloud API (${input.scanType} scan creation).`,
            `Target resources: application ${appName || "Unknown"} (${input.appId}), new scan record${input.scanName ? ` named ${input.scanName}` : ""}.`,
            `Personal scan: ${input.personalScan ? "yes" : "no"}.`,
            `DAST target URL: ${input.scanUrl}.`,
            `Network mode: ${input.networkType}${input.networkType === "private" ? ` (PresenceId: ${input.presenceId})` : ""}.`,
            `Authentication mode: ${input.loginMethod}.`,
            `Local filesystem read/write: none required for DAST packaging flow.`,
            `Other changes: starts remote DAST execution; may update remote scan state while waiting when wait=true.`,
        ].join("\n");
    }
    return [
        `Tool: create-scan`,
        `Target environment: AppScan on Cloud API (${input.scanType} scan creation).`,
        `Target resources: application ${appName || "Unknown"} (${input.appId}), new scan record${input.scanName ? ` named ${input.scanName}` : ""}.`,
        `Personal scan: ${input.personalScan ? "yes" : "no"}.`,
        `Local filesystem read/write: target directory ${input.targetDir}; config file ${effectiveConfig}; working directory ${effectiveWorkDir}.`,
        `Potential deletions: generated local artifacts and generated appscan-config.xml are deleted when conserveFiles=false.`,
        `Build execution: ${input.buildBeforeScan ? "npm run build will execute in target directory" : "no build command will be executed"}.`,
        `Other changes: uploads IRX artifact to AppScan and starts remote scan execution; may update remote scan state while waiting when wait=true.`,
    ].join("\n");
}
function formatAvailableApps(apps) {
    return apps
        .slice(0, 10)
        .map((app) => `- ${app.Name || "Unknown"} (ID: ${app.Id}, Type: ${app.AppType || "Unknown"})`)
        .join("\n");
}
async function getAppsForSelection() {
    const appsResponse = await makeAppScanRequest("/api/v4/Apps?$top=10");
    return Array.isArray(appsResponse) ? appsResponse : appsResponse.Items;
}
async function findAppById(appId) {
    // OData v4 requires raw GUIDs without the guid'' prefix or single quotes
    const filter = encodeURIComponent(`Id eq ${appId}`);
    const response = await makeAppScanRequest(`/api/v4/Apps?$filter=${filter}&$top=1`);
    const apps = Array.isArray(response) ? response : response.Items;
    return apps.length > 0 ? apps[0] : null;
}

import { createHash, randomUUID } from "crypto";
import * as fs from "fs";
import * as path from "path";
const TOKEN_TTL_MS = 5 * 60 * 1000;
const approvalTokens = new Map();
const auditLogFile = path.join(process.cwd(), "appscan-sensitive-tools-audit.log");
function stableNormalize(value) {
    if (Array.isArray(value)) {
        return value.map(stableNormalize);
    }
    if (value && typeof value === "object") {
        const obj = value;
        const sortedKeys = Object.keys(obj).sort();
        const result = {};
        for (const key of sortedKeys) {
            result[key] = stableNormalize(obj[key]);
        }
        return result;
    }
    return value;
}
function hashArgs(args) {
    const normalized = stableNormalize(args);
    return createHash("sha256").update(JSON.stringify(normalized)).digest("hex");
}
function redactSecrets(value, parentKey = "") {
    const sensitiveKey = /(token|password|secret|api[_-]?key|key|auth|credential)/i.test(parentKey);
    if (value === null || value === undefined) {
        return value;
    }
    if (Array.isArray(value)) {
        return value.map((item) => redactSecrets(item, parentKey));
    }
    if (typeof value === "object") {
        const input = value;
        const output = {};
        for (const [k, v] of Object.entries(input)) {
            output[k] = redactSecrets(v, k);
        }
        return output;
    }
    if (sensitiveKey) {
        return "[REDACTED]";
    }
    if (typeof value === "string" && /(?:[A-Za-z0-9_-]{20,}|[0-9a-fA-F]{32,})/.test(value) && sensitiveKey) {
        return "[REDACTED]";
    }
    return value;
}
function cleanupExpiredTokens(now = Date.now()) {
    for (const [token, data] of approvalTokens.entries()) {
        if (now - data.createdAt > TOKEN_TTL_MS) {
            approvalTokens.delete(token);
        }
    }
}
async function appendAuditLog(toolName, args, approvalPath, outcome) {
    try {
        const line = JSON.stringify({
            timestamp: new Date().toISOString(),
            toolName,
            arguments: redactSecrets(args),
            approvalPath,
            outcome,
        });
        await fs.promises.appendFile(auditLogFile, `${line}\n`, "utf-8");
    }
    catch (error) {
        // Diagnostics must not go to stdout on stdio MCP servers.
        console.error(`[approval] failed to append audit log: ${error instanceof Error ? error.message : String(error)}`);
    }
}
function makePreviewText(toolName, summary, token) {
    return [
        `Approval required for sensitive tool: ${toolName}`,
        "",
        "Preview of pending action:",
        summary,
        "",
        `One-time confirmToken (expires in 5 minutes): ${token}`,
        "Do not execute yet.",
        "Assistant instruction: show this preview to the user and get explicit agreement, then call the same tool again with the same arguments plus confirmToken.",
    ].join("\n");
}
export async function requireApproval(server, toolName, args, summary, options = {}) {
    cleanupExpiredTokens();
    const { confirmToken, ...argsWithoutToken } = args;
    try {
        if (toolName === "create-scan" && process.env.APPSCAN_CREATE_SCAN_SKIP_ELICITATION === "true") {
            await appendAuditLog(toolName, argsWithoutToken, "automation", "approved");
            return { approved: true };
        }
        const supportsElicitation = Boolean(server.server?.getClientCapabilities?.()?.elicitation);
        if (supportsElicitation) {
            try {
                const approveLabel = options.approveLabel || "proceed";
                const rejectLabel = options.rejectLabel || "cancel";
                const elicitationMessage = options.elicitationMessage ||
                    `Do you want to proceed with the ${toolName} action?`;
                const elicitation = await server.server.elicitInput({
                    message: [
                        elicitationMessage,
                        "",
                        `Approve: ${approveLabel}.`,
                        `Reject: ${rejectLabel}.`,
                    ].join("\n"),
                    requestedSchema: {
                        type: "object",
                        properties: {
                            Decision: {
                                type: "string",
                                enum: ["Approve", "Reject"],
                                description: `Approve: ${approveLabel}. Reject: ${rejectLabel}.`,
                            },
                        },
                        required: ["Decision"],
                        additionalProperties: false,
                    },
                });
                const decision = elicitation?.content?.Decision;
                const accepted = elicitation?.action === "accept" &&
                    typeof decision === "string" &&
                    decision.toLowerCase() === "approve";
                if (!accepted) {
                    const outcome = elicitation?.action === "decline" || (typeof decision === "string" && decision.toLowerCase() === "reject")
                        ? "declined"
                        : "cancelled";
                    await appendAuditLog(toolName, argsWithoutToken, "elicitation", outcome);
                    return { approved: false, responseText: "Cancelled by user." };
                }
                await appendAuditLog(toolName, argsWithoutToken, "elicitation", "approved");
                return { approved: true };
            }
            catch {
                // Fall back to preview-confirm if elicitation support was claimed but failed.
            }
        }
        const approvalPath = "preview-confirm";
        const token = typeof confirmToken === "string" ? confirmToken : undefined;
        const argsHash = hashArgs(argsWithoutToken);
        if (!token) {
            const newToken = randomUUID();
            approvalTokens.set(newToken, { toolName, argsHash, createdAt: Date.now() });
            await appendAuditLog(toolName, argsWithoutToken, approvalPath, "cancelled");
            return { approved: false, responseText: makePreviewText(toolName, summary, newToken) };
        }
        const tokenData = approvalTokens.get(token);
        approvalTokens.delete(token);
        if (!tokenData) {
            await appendAuditLog(toolName, argsWithoutToken, approvalPath, "invalid-token");
            return {
                approved: false,
                responseText: "Invalid or already-used confirmToken. Request a fresh preview and token before trying again.",
            };
        }
        const isExpired = Date.now() - tokenData.createdAt > TOKEN_TTL_MS;
        if (isExpired) {
            await appendAuditLog(toolName, argsWithoutToken, approvalPath, "expired-token");
            return {
                approved: false,
                responseText: "confirmToken has expired. Request a fresh preview and token before trying again.",
            };
        }
        if (tokenData.toolName !== toolName || tokenData.argsHash !== argsHash) {
            await appendAuditLog(toolName, argsWithoutToken, approvalPath, "invalid-token");
            return {
                approved: false,
                responseText: "confirmToken does not match this tool call or arguments. Request a fresh preview and token before trying again.",
            };
        }
        await appendAuditLog(toolName, argsWithoutToken, approvalPath, "approved");
        return { approved: true };
    }
    catch {
        await appendAuditLog(toolName, argsWithoutToken, "preview-confirm", "error");
        return {
            approved: false,
            responseText: "Cancelled by user.",
        };
    }
}
