export type AppScanProvider = "remote" | "local";

export interface ConnectorConfig {
  provider: AppScanProvider;
  requestTimeoutMs: number;
  remote?: {
    endpoint: string;
    apiKey?: string;
  };
  local?: {
    command: string;
    args: string[];
    serverPath: string;
    apiBase: string;
    keyId?: string;
    keySecret?: string;
  };
}

export type Environment = Record<string, string | undefined>;

export function loadConfig(
  env: Environment = process.env,
): ConnectorConfig {
  const provider = (env.APP_SCAN_PROVIDER ?? "remote")
    .trim()
    .toLowerCase();

  if (provider !== "remote" && provider !== "local") {
    throw new Error(
      'APP_SCAN_PROVIDER must be either "remote" or "local".',
    );
  }

  const timeout = Number(env.APP_SCAN_REQUEST_TIMEOUT_MS ?? "30000");

  if (!Number.isSafeInteger(timeout) || timeout < 1 || timeout > 300000) {
    throw new Error(
      "APP_SCAN_REQUEST_TIMEOUT_MS must be an integer between 1 and 300000.",
    );
  }

  if (provider === "remote") {
    const endpoint = env.APP_SCAN_REMOTE_ENDPOINT?.trim();

    if (!endpoint) {
      throw new Error(
        "APP_SCAN_REMOTE_ENDPOINT is required for remote mode.",
      );
    }

    let parsed: URL;

    try {
      parsed = new URL(endpoint);
    } catch {
      throw new Error("APP_SCAN_REMOTE_ENDPOINT must be a valid URL.");
    }

    if (parsed.protocol !== "https:" && parsed.protocol !== "http:") {
      throw new Error("Remote endpoint must use HTTP or HTTPS.");
    }

    return {
      provider,
      requestTimeoutMs: timeout,
      remote: {
        endpoint: parsed.toString(),
        apiKey: env.APP_SCAN_API_KEY?.trim() || undefined,
      },
    };
  }

  const serverPath = env.APP_SCAN_LOCAL_SERVER_PATH?.trim();

  if (!serverPath) {
    throw new Error(
      "APP_SCAN_LOCAL_SERVER_PATH is required for local mode.",
    );
  }

  return {
    provider,
    requestTimeoutMs: timeout,
    local: {
      command: env.APP_SCAN_LOCAL_COMMAND?.trim() || "node",
      args: [serverPath],
      serverPath,
      apiBase:
        env.APPSCAN_API_BASE?.trim() || "https://cloud.appscan.com",
      keyId: env.APPSCAN_KEY_ID?.trim() || undefined,
      keySecret: env.APPSCAN_KEY_SECRET?.trim() || undefined,
    },
  };
}

import { randomUUID } from "node:crypto";
import type { AppScanService } from "./appscan-service.js";

export interface PreparedScan {
  appId: string;
  targetDir: string;
  scanType: "SAST" | "SCA";
  scanName?: string;
  comment?: string;
}

export interface ScanWorkflowResult {
  token?: string;
  text: string;
  isError?: boolean;
}

export class ScanWorkflow {
  private readonly pending = new Map<string, PreparedScan>();

  constructor(private readonly service: AppScanService) {}

  prepare(input: PreparedScan): ScanWorkflowResult {
    const token = randomUUID();
    this.pending.set(token, { ...input });

    return {
      token,
      text: [
        "Scan prepared; no scan has been started.",
        `Confirmation token: ${token}`,
        `Application ID: ${input.appId}`,
        `Target directory: ${input.targetDir}`,
        `Scan type: ${input.scanType}`,
        `Scan name: ${input.scanName ?? "(generated by AppScan)"}`,
        "",
        "Before execution, explain that source artifacts may be packaged and uploaded to AppScan, and that the underlying tool may generate configuration files or run a build.",
        "Obtain explicit user confirmation for these exact parameters before execution.",
      ].join("\n"),
    };
  }

  async execute(token: string): Promise<ScanWorkflowResult> {
    const prepared = this.pending.get(token);

    if (!prepared) {
      return {
        text: "No pending scan matches this token. Prepare the request again and obtain user confirmation.",
        isError: true,
      };
    }

    // Consume before invoking AppScan, preventing token reuse after failure.
    this.pending.delete(token);

    const result = await this.service.callTool("create-scan", {
      ...prepared,
      approved: true,
    });

    const text = result.content
      .map((item) => item.text ?? "")
      .filter(Boolean)
      .join("\n");

    return {
      text: text || "The AppScan tool returned no text content.",
      ...(result.isError ? { isError: true } : {}),
    };
  }
}

import { Client } from "@modelcontextprotocol/sdk/client/index.js";
import { StdioClientTransport } from "@modelcontextprotocol/sdk/client/stdio.js";
import type { Tool } from "@modelcontextprotocol/sdk/types.js";

import type {
  AppScanMcpConnector,
  McpTool,
  McpToolCallResult,
} from "./connector.js";

export interface LocalStdioConnectorOptions {
  command: string;
  args: string[];
  env?: Record<string, string>;
  cwd?: string;
  requestTimeoutMs?: number;
}

export class LocalStdioConnector implements AppScanMcpConnector {
  private readonly client: Client;
  private readonly transport: StdioClientTransport;
  private readonly requestTimeoutMs: number;
  private connected = false;

  constructor(options: LocalStdioConnectorOptions) {
    this.requestTimeoutMs = options.requestTimeoutMs ?? 30000;

    this.client = new Client({
      name: "hcl-appscan-connector-for-claude",
      version: "0.1.0",
    });

    this.transport = new StdioClientTransport({
      command: options.command,
      args: options.args,
      env: options.env,
      cwd: options.cwd,
      stderr: "inherit",
    });
  }

  async connect(): Promise<void> {
    if (this.connected) {
      return;
    }

    await this.client.connect(this.transport);
    this.connected = true;
  }

  async listTools(): Promise<McpTool[]> {
    this.ensureConnected();

    const result = await this.client.listTools(
      {},
      { timeout: this.requestTimeoutMs },
    );

    return result.tools.map((tool: Tool) => ({
      name: tool.name,
      description: tool.description,
      inputSchema: tool.inputSchema as Record<string, unknown>,
    }));
  }

  async callTool(
    name: string,
    args: Record<string, unknown>,
  ): Promise<McpToolCallResult> {
    this.ensureConnected();

    const result: unknown = await this.client.callTool(
      {
        name,
        arguments: args,
      },
      undefined,
      { timeout: this.requestTimeoutMs },
    );

    if (
      typeof result !== "object" ||
      result === null ||
      !("content" in result) ||
      !Array.isArray(result.content)
    ) {
      throw new Error(
        `MCP tool "${name}" returned an unsupported result format.`,
      );
    }

    const content: McpToolCallResult["content"] = result.content.map(
      (item: unknown) => {
        if (typeof item !== "object" || item === null || !("type" in item)) {
          throw new Error(
            `MCP tool "${name}" returned an invalid content item.`,
          );
        }

        const type = item.type;

        if (typeof type !== "string") {
          throw new Error(
            `MCP tool "${name}" returned a content item with an invalid type.`,
          );
        }

        if ("text" in item && typeof item.text === "string") {
          return { type, text: item.text };
        }

        return { type };
      },
    );

    return {
      content,
      isError:
        "isError" in result && typeof result.isError === "boolean"
          ? result.isError
          : undefined,
    };
  }

  async disconnect(): Promise<void> {
    if (!this.connected) {
      return;
    }

    try {
      await this.client.close();
    } finally {
      this.connected = false;
    }
  }

  private ensureConnected(): void {
    if (!this.connected) {
      throw new Error("Local AppScan MCP connector is not connected.");
    }
  }
}
