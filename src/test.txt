// AppScan API Configuration
const APPSCAN_API_BASE = process.env.APPSCAN_API_BASE || "https://cloud.appscan.com";
const APPSCAN_KEY_ID = process.env.APPSCAN_KEY_ID;
const APPSCAN_KEY_SECRET = process.env.APPSCAN_KEY_SECRET;
function getDirectApiKeyHeader() {
    if (!APPSCAN_KEY_ID || !APPSCAN_KEY_SECRET) {
        throw new Error("AppScan credentials are not configured. Set APPSCAN_KEY_ID and APPSCAN_KEY_SECRET.");
    }
    return `${APPSCAN_KEY_ID}:${APPSCAN_KEY_SECRET}`;
}
// Debug: Log environment variable status (values are masked for security)
console.error(`[DEBUG] APPSCAN_API_BASE: ${APPSCAN_API_BASE}`);
console.error(`[DEBUG] APPSCAN_KEY_ID: ${APPSCAN_KEY_ID ? `SET (${APPSCAN_KEY_ID.substring(0, 4)}...)` : 'NOT SET'}`);
