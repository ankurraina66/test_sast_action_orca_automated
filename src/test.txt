it("calls an inactive provider without switching the active provider", async () => {
    const activeConnector = createMockConnector();
    const localConnector = createMockConnector(["get-apps"]);

    const factory = vi.fn(() => localConnector);

    const service = createService(
        activeConnector,
        "remote",
        factory,
    );

    await service.callProviderTool(
        "local",
        "get-apps",
        { top: 10 },
    );

    expect(factory).toHaveBeenCalledWith("local");

    expect(localConnector.connect).toHaveBeenCalledTimes(1);

    expect(localConnector.callTool).toHaveBeenCalledWith(
        "get-apps",
        { top: 10 },
    );

    expect(localConnector.disconnect).toHaveBeenCalledTimes(1);

    expect(service.getProvider()).toBe("remote");

    expect(activeConnector.callTool).not.toHaveBeenCalled();
});

it("reuses the active provider connector without creating a temporary connector", async () => {
    const activeConnector = createMockConnector();

    const factory = vi.fn(() => createMockConnector());

    const service = createService(
        activeConnector,
        "local",
        factory,
    );

    await service.callProviderTool(
        "local",
        "get-apps",
        { top: 5 },
    );

    expect(activeConnector.callTool).toHaveBeenCalledWith(
        "get-apps",
        { top: 5 },
    );

    expect(factory).not.toHaveBeenCalled();

    expect(service.getProvider()).toBe("local");
});
