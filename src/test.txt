
import { createAppScanService } from "../../src/services/appscan-service-factory.js";

async function main(): Promise<void> {
  const service = createAppScanService();

  try {
    await service.connect();

    console.log("Retrieving findings for the selected SAST scan...");

    const result = await service.callTool("get-scan-issues", {
      scanId: "e3ddcad9-430c-481f-88dd-0bb2f2fb4793",
      top: 5,
    });

    for (const item of result.content) {
      if (item.text) {
        console.log(item.text);
      }
    }

    if (result.isError) {
      throw new Error("AppScan returned an error while retrieving findings.");
    }

    console.log("Finding retrieval completed.");
  } finally {
    await service.disconnect();
  }
}

main().catch((error: unknown) => {
  console.error(
    "Finding retrieval failed:",
    error instanceof Error ? error.message : "Unknown error",
  );
  process.exitCode = 1;
});
