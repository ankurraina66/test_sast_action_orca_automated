@'
import { createAppScanService } from "./src/services/appscan-service-factory.ts";

const service = createAppScanService();

try {
  await service.connect();
  const tools = await service.listTools();

  for (const name of ["get-apps", "get-scans"]) {
    const tool = tools.find((item) => item.name === name);
    console.log(`\n${name} input schema:`);
    console.log(JSON.stringify(tool?.inputSchema ?? "NOT FOUND", null, 2));
  }
} finally {
  await service.disconnect();
}
'@ | Set-Content .\scripts\inspect-appscan-schemas.ts

npx tsx .\scripts\inspect-appscan-schemas.ts
