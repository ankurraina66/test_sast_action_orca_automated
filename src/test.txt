export function normalizeCapabilityArguments(
    capability: string,
    provider: AppScanProvider,
    request: CapabilityRequest,
): Record<string, unknown> {
    const {
        capability: _capability,
        appId,
        scanId,
        issueId,
        ...common
    } = request;

    switch (capability) {
        case "scan-issues": {
            const resolvedScanId = scanId ?? request.scopeId;

            if (!resolvedScanId) {
                throw new Error(
                    "scanId or scopeId is required for scan-issues.",
                );
            }

            if (provider === "remote") {
                return {
                    ...common,
                    scope: "Scan",
                    scopeId: resolvedScanId,
                };
            }

            return {
                ...common,
                scanId: resolvedScanId,
            };
        }

        case "issues": {
            const resolvedAppId = appId ?? request.scopeId;

            if (provider === "local" && !resolvedAppId) {
                throw new Error(
                    "appId or scopeId is required for Local issues.",
                );
            }

            if (provider === "local") {
                return {
                    ...common,
                    appId: resolvedAppId,
                };
            }

            return {
                ...common,
                ...(request.scope ? { scope: request.scope } : {}),
                ...(resolvedAppId ? { scopeId: resolvedAppId } : {}),
            };
        }

        case "issue-details":
            if (!issueId) {
                throw new Error(
                    "issueId is required for issue-details.",
                );
            }

            return { issueId };

        default:
            return common;
    }
}
