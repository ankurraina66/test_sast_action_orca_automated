import { createAppScanService } from "../../src/services/appscan-service-factory.js";

async function main(): Promise<void> {
  const service = createAppScanService();

  try {
    console.log("Connecting to local AppScan MCP...");

    await service.connect();

    console.log("Connection established. Discovering tools...");

    const tools = await service.listTools();

    console.log(`Discovered ${tools.length} tool(s):`);

    for (const tool of tools) {
      console.log(`- ${tool.name}`);
    }

    console.log("Local MCP connection and tool discovery succeeded.");
  } finally {
    await service.disconnect();
  }
}

main().catch((error: unknown) => {
  console.error(
    "Local MCP integration test failed:",
    error instanceof Error ? error.message : "Unknown error",
  );
  process.exitCode = 1;
});
