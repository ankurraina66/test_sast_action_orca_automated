
function jsonSchemaToZod(schema: unknown): z.ZodTypeAny {
    if (!schema || typeof schema !== "object") {
        return z.unknown();
    }

    const definition = schema as Record<string, unknown>;

    if (definition.const !== undefined) {
        return z.literal(definition.const as string | number | boolean);
    }

    if (Array.isArray(definition.enum)) {
        const values = definition.enum;

        if (
            values.length > 0 &&
            values.every((value) => typeof value === "string")
        ) {
            return z.enum(values as [string, ...string[]]);
        }

        if (values.length > 0) {
            return z.union(
                values.map((value) => z.literal(value)) as [
                    z.ZodTypeAny,
                    z.ZodTypeAny,
                    ...z.ZodTypeAny[],
                ],
            );
        }
    }

    if (Array.isArray(definition.anyOf) || Array.isArray(definition.oneOf)) {
        const variants = (definition.anyOf ?? definition.oneOf) as unknown[];

        if (variants.length >= 2) {
            return z.union(
                variants.map(jsonSchemaToZod) as [
                    z.ZodTypeAny,
                    z.ZodTypeAny,
                    ...z.ZodTypeAny[],
                ],
            );
        }
    }

    const type = definition.type;

    let result: z.ZodTypeAny;

    if (type === "string") {
        let value = z.string();

        if (typeof definition.minLength === "number") {
            value = value.min(definition.minLength);
        }

        if (typeof definition.maxLength === "number") {
            value = value.max(definition.maxLength);
        }

        if (typeof definition.pattern === "string") {
            value = value.regex(new RegExp(definition.pattern));
        }

        result = value;
    } else if (type === "integer") {
        let value = z.number().int();

        if (typeof definition.minimum === "number") {
            value = value.min(definition.minimum);
        }

        if (typeof definition.maximum === "number") {
            value = value.max(definition.maximum);
        }

        result = value;
    } else if (type === "number") {
        let value = z.number();

        if (typeof definition.minimum === "number") {
            value = value.min(definition.minimum);
        }

        if (typeof definition.maximum === "number") {
            value = value.max(definition.maximum);
        }

        result = value;
    } else if (type === "boolean") {
        result = z.boolean();
    } else if (type === "array") {
        result = z.array(
            jsonSchemaToZod(definition.items),
        );
    } else if (type === "object" || definition.properties) {
        const properties = (
            definition.properties ?? {}
        ) as Record<string, unknown>;

        const required = new Set(
            Array.isArray(definition.required)
                ? definition.required.filter(
                    (item): item is string => typeof item === "string",
                )
                : [],
        );

        const shape: Record<string, z.ZodTypeAny> = {};

        for (const [key, propertySchema] of Object.entries(properties)) {
            const property = jsonSchemaToZod(propertySchema);

            shape[key] = required.has(key)
                ? property
                : property.optional();
        }

        result = z.object(shape);
    } else {
        // Safely handle schemas with unsupported JSON Schema constructs.
        result = z.unknown();
    }

    if (definition.nullable === true) {
        result = result.nullable();
    }

    if (definition.default !== undefined) {
        result = result.default(definition.default);
    }

    return result;
}



    // Dynamically register individual read-only AppScan capabilities.
    try {
        const { local, remote } =
            await service.compareProviderTools();

        const catalogue = buildToolCatalogue(local, remote);

        for (const item of catalogue) {
            // Do not expose write or mixed operations through this
            // read-only dynamic registration.
            if (item.risk !== "read") {
                continue;
            }

            const providers = [
                item.local ? "Local" : undefined,
                item.remote ? "Remote" : undefined,
            ].filter(Boolean).join(", ");

            const selectedTool = item.local ?? item.remote;

            if (!selectedTool) {
                continue;
            }

            const inputSchema = jsonSchemaToZod(
                selectedTool.inputSchema,
            );

            server.registerTool(
                `appscan-${item.capability}`,
                {
                    description: [
                        `Execute the unified AppScan capability: ${item.capability}.`,
                        `Available providers: ${providers}.`,
                        "Local is preferred when available; otherwise Remote is used.",
                        "This operation is read-only and does not change the active provider.",
                        `Backend tool description: ${selectedTool.description ?? "No description provided."}`,
                    ].join(" "),
                    inputSchema,
                },
                async (args) => {
                    try {
                        const result = await service.callCapability(
                            item.capability,
                            args as Record<string, unknown>,
                        );

                        return textResult(result);
                    } catch (error) {
                        return messageResult(
                            error instanceof Error
                                ? error.message
                                : `AppScan capability '${item.capability}' failed.`,
                            true,
                        );
                    }
                },
            );
        }

        console.error(
            `[AppScan] Registered ${catalogue.filter(
                (item) => item.risk === "read",
            ).length} unified read-only capability tools.`,
        );
    } catch (error) {
        console.error(
            "[AppScan] Could not register dynamic capability tools:",
            error instanceof Error ? error.message : "Unknown error",
        );

        throw error;
    }
