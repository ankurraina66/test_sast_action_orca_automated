/**
 * Tool: get-scans
 * Get list of scans
 */
server.registerTool("get-scans", {
    description: "Get list of scans in AppScan",
    inputSchema: {
        filter: z.string().optional().describe("OData filter expression (optional)"),
        top: z.number().optional().describe("Number of records to return (optional)"),
    },
}, async ({ filter, top }) => {
    try {
        let endpoint = "/api/v4/Scans";
        const params = new URLSearchParams();
        if (filter)
            params.append("$filter", filter);
        if (top)
            params.append("$top", top.toString());
        if (params.toString()) {
            endpoint += `?${params.toString()}`;
        }
        const response = await makeAppScanRequest(endpoint);
        const scans = response.Items || response || [];
        if (scans.length === 0) {
            return {
                content: [
                    {
                        type: "text",
                        text: "No scans found",
                    },
                ],
            };
        }
        const formattedScans = scans.map((scan) => [
            `Name: ${scan.Name || "Unknown"}`,
            `ID: ${scan.Id || "Unknown"}`,
            `Type: ${scan.ScanType || "Unknown"}`,
            `Status: ${scan.LatestExecution?.Status || "Unknown"}`,
            `Created: ${scan.Created || "Unknown"}`,
            `Risk: ${scan.LatestExecution?.Risk || "Unknown"}`,
            "---",
        ].join("\n"));
        return {
            content: [
                {
                    type: "text",
                    text: `Found ${scans.length} scan(s):\n\n${formattedScans.join("\n")}`,
                },
            ],
        };
    }
    catch (error) {
        return {
            content: [
                {
                    type: "text",
                    text: `Error fetching scans: ${error instanceof Error ? error.message : String(error)}`,
                },
            ],
            isError: true,
        };
    }
});
