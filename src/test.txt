import { createAppScanService } from "../../src/services/appscan-service-factory.js";

async function main(): Promise<void> {
  const service = createAppScanService();

  try {
    await service.connect();

    const tools = await service.listTools();
    const tool = tools.find((item) => item.name === "get-issues");

    if (!tool) {
      throw new Error("The get-issues tool is not available.");
    }

    console.log("get-issues input schema:");
    console.log(JSON.stringify(tool.inputSchema, null, 2));
  } finally {
    await service.disconnect();
  }
}

main().catch((error: unknown) => {
  console.error(
    "get-issues schema inspection failed:",
    error instanceof Error ? error.message : "Unknown error",
  );
  process.exitCode = 1;
});

import { createAppScanService } from "../../src/services/appscan-service-factory.js";

async function main(): Promise<void> {
  const service = createAppScanService();

  try {
    await service.connect();

    console.log("Retrieving up to 5 AppScan applications...");

    const result = await service.callTool("get-apps", { top: 5 });

    for (const item of result.content) {
      if (item.text) {
        console.log(item.text);
      }
    }

    if (result.isError) {
      throw new Error("AppScan returned an error while retrieving applications.");
    }

    console.log("Application retrieval completed.");
  } finally {
    await service.disconnect();
  }
}

main().catch((error: unknown) => {
  console.error(
    "Application retrieval failed:",
    error instanceof Error ? error.message : "Unknown error",
  );
  process.exitCode = 1;
});
