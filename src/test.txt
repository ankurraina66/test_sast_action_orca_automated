

    it("checks capability and executes using the same locked provider operation", async () => {
        const connector = createMockConnector(["create-scan"]);
        const service = createService(connector, "local");

        await expect(
            service.callFirstAvailableToolIfSupported(
                ["create-scan", "create_scan"],
                { approved: true },
                "local",
            ),
        ).resolves.toMatchObject({
            content: [{ type: "text", text: "Success" }],
            isError: false,
        });

        expect(connector.listTools).toHaveBeenCalledTimes(1);
        expect(connector.callTool).toHaveBeenCalledWith("create-scan", {
            approved: true,
        });
    });

    it("does not execute when the expected provider is not active", async () => {
        const connector = createMockConnector(["create-scan"]);
        const service = createService(connector, "remote");

        await expect(
            service.callFirstAvailableToolIfSupported(
                ["create-scan"],
                { approved: true },
                "local",
            ),
        ).rejects.toThrow("Provider mismatch");

        expect(connector.callTool).not.toHaveBeenCalled();
    });

    it("does not execute when the requested tool is unavailable", async () => {
        const connector = createMockConnector(["get-apps"]);
        const service = createService(connector, "remote");

        await expect(
            service.callFirstAvailableToolIfSupported(
                ["create-scan", "create_scan"],
                { approved: true },
                "remote",
            ),
        ).rejects.toThrow("does not support this operation");

        expect(connector.callTool).not.toHaveBeenCalled();
    });
