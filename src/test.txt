const reportScopeSchema = z.enum(["Application", "Scan", "ScanExecution"]);

const commonReportFilterSchema = {
    odataFilter: z.string().optional().describe("OData filter expression to limit issues included in the report."),
    applyPolicies: applyPoliciesSchema.optional().describe("Which policies to apply for filtering (default: None)."),

function formatApprovalScope(scope) {
    return scope.replace(/([a-z])([A-Z])/g, "$1 $2").toLowerCase();
}
const contentTypeExtensions = {
    "text/html": ".html",
    "application/pdf": ".pdf",
    "application/xml": ".xml",
    "text/xml": ".xml",
    "text/csv": ".csv",
    "application/json": ".json",
    "application/zip": ".zip",
};
function defaultReportFileName(reportId, fileName, contentType) {
    const sanitized = fileName ? sanitizeReportFileName(fileName) : "";
    if (sanitized) {
        return sanitized;
    }
    const extension = contentTypeExtensions[contentType.split(";")[0].trim().toLowerCase()] ?? ".bin";
    return `appscan-report-${reportId}${extension}`;
}
function buildReportScopePayload(input) {
    const payload = {};
    if (input.odataFilter)
        payload.OdataFilter = input.odataFilter;
    if (input.applyPolicies)
        payload.ApplyPolicies = input.applyPolicies;
    if (input.selectPolicyIds && input.selectPolicyIds.length > 0)
        payload.SelectPolicyIds = input.selectPolicyIds;
    return payload;
}
function formatReportJobResponse(response) {
    const id = typeof response.Id === "string" ? response.Id : "Unknown";
    const name = typeof response.Name === "string" ? response.Name : "Unknown";
    const status = typeof response.Status === "string" ? response.Status : "Pending";
    const progress = typeof response.Progress === "number" ? response.Progress : "Unknown";
    const validUntil = typeof response.ValidUntil === "string" ? response.ValidUntil : "Unknown";
    const reportFileType = typeof response.ReportFileType === "string" ? response.ReportFileType : "Unknown";
    const downloadLink = typeof response.DownloadLink === "string" ? response.DownloadLink : "Not available yet";
    return [
        `ID: ${id}`,
        `Name: ${name}`,
        `Status: ${status}`,
        `Progress: ${progress}`,
        `Valid Until: ${validUntil}`,
        `Report File Type: ${reportFileType}`,
        `Download Link: ${downloadLink}`,
    ].join("\n");
}

    server.registerTool("get-apps", {
        description: getToolDescription("get-apps"),
        inputSchema: {
            filter: z.string().optional().describe("OData filter expression (optional)"),
            top: z.number().optional().describe("Number of records to return (optional)"),
        },
    }, async ({ filter, top }) => {
        try {
            let endpoint = "/api/v4/Apps";
            const params = new URLSearchParams();
            if (filter)
                params.append("$filter", filter);
            if (top)
                params.append("$top", top.toString());
            params.append("$count", "true");
            if (params.toString()) {
                endpoint += `?${params.toString()}`;
            }
            const response = await makeAppScanRequest(endpoint);
            const apps = Array.isArray(response) ? response : response.Items;
            const totalCount = !Array.isArray(response) && typeof response.Count === "number" ? response.Count : undefined;
            const completeness = totalCount === undefined ? "unknown" : totalCount === apps.length ? "yes" : "no";
            const uniqueExplicitMatch = canSelectUniqueCandidate(apps.length, totalCount, Boolean(filter?.trim()));
            const summary = [
                apps.length === 0
                    ? `No AppScan applications were found${filter ? ` for selector: ${filter}` : ""}.`
                    : `Found ${apps.length} application(s).`,
                `Matching total: ${totalCount ?? "unknown"}. Complete result: ${completeness}.`,
                `Unique match for explicit filter: ${uniqueExplicitMatch ? "yes" : "no"}.`,
                `Filter: ${filter || "(none)"}`,
            ];
            let availableApps = apps;
            if (apps.length === 0 && filter?.trim()) {
                const availableResponse = await makeAppScanRequest("/api/v4/Apps?$top=10&$count=true");
                availableApps = Array.isArray(availableResponse) ? availableResponse : availableResponse.Items;
                const availableCount = !Array.isArray(availableResponse) ? availableResponse.Count : undefined;
                summary.push("", "The choices below are other existing applications, not matches for the requested selector.", `Showing ${availableApps.length} application(s). Available total: ${availableCount ?? "unknown"}. This may be a partial list; search by name or ID for other applications.`);
            }
            const formattedApps = availableApps.map((app) => [
                `Name: ${app.Name || "Unknown"}`,
                `ID: ${app.Id || "Unknown"}`,
                `Type: ${app.AppType || "Unknown"}`,
                `Created: ${app.Created || "Unknown"}`,
                `Risk: ${app.Risk || "Unknown"}`,
                "---",
            ].join("\n"));
            return {
                content: [
                    {
                        type: "text",
                        text: uniqueExplicitMatch
                            ? [...summary, "", formattedApps.join("\n")].join("\n")
                            : formatAppResolutionMessage(summary.join("\n"), formattedApps.join("\n")),
                    },
                ],
            };
        }
        catch (error) {
            return {
                content: [
                    {
                        type: "text",
                        text: `Error fetching apps: ${error instanceof Error ? error.message : String(error)}`,
                    },
                ],
                isError: true,
            };
        }

            filter: z.string().optional().describe("OData filter expression."),
            select: z.string().optional().describe("OData select expression."),
            orderby: z.string().optional().describe("OData orderby expression."),
            expand: z.string().optional().describe("OData expand expression."),
            count: z.boolean().optional().describe("Include total count in response."),
            apply: z.string().optional().describe("OData apply expression."),
        },
    }, async ({ scope, scopeId, applyPolicies, selectPolicyIds, locale, top, skip, filter, select, orderby, expand, count, apply }) => {
        try {
            if (applyPolicies === "Select" && (!selectPolicyIds || selectPolicyIds.length === 0)) {
                return {
                    content: [
                        {
                            type: "text",
                            text: "I need more details before I can get fix groups:\n- selectPolicyIds: required when applyPolicies is 'Select'",
                        },
                    ],
                    isError: true,
                };
            }
            let endpoint = `/api/v4/FixGroups/${scope}/${scopeId}`;
            const params = new URLSearchParams();
            if (applyPolicies)
                params.append("applyPolicies", applyPolicies);
            if (locale)
                params.append("locale", locale);
            if (typeof top === "number")
                params.append("$top", top.toString());
            if (typeof skip === "number")
                params.append("$skip", skip.toString());
            if (filter)
                params.append("$filter", filter);
            if (select)
                params.append("$select", select);
            if (orderby)
                params.append("$orderby", orderby);
            if (expand)
                params.append("$expand", expand);
            if (typeof count === "boolean")
                params.append("$count", count.toString());
            if (apply)
                params.append("$apply", apply);
            if (applyPolicies === "Select" && selectPolicyIds) {
                for (const policyId of selectPolicyIds) {
                    params.append("selectPolicyIds", policyId);
                }
            }
            if (params.toString()) {
                endpoint += `?${params.toString()}`;
            }
            const response = await makeAppScanRequest(endpoint);
            const items = response.Items;
            const fixGroups = Array.isArray(items) ? items : [];

                    content: [

                            type: "text",
                            text: "No fix groups found for this scope",
                        },
                    ],
                };

                `Language: ${typeof fixGroup.Language === "string" ? fixGroup.Language : "Unknown"}`,
                "---",
            ].join("\n"));

                content: [
                    {
                        type: "text",
                        text: `Found ${fixGroups.length} fix group(s) (Count: ${totalCount}):\n\n${formattedFixGroups.join("\n")}`,

        },
    }, async ({ scope, scopeId, appId, scanId, issueId, status, comment, externalId, odataFilter, applyPolicies, selectPolicyIds, confirmToken, allowDuplicate }) => {
        try {
            const targetResolution = resolveIssueUpdateTarget({ scope, scopeId, appId, scanId });
            if (!targetResolution.ok) {
                return {
                    content: [{ type: "text", text: targetResolution.error }],
                    isError: true,
                };
            }
            const { scope: resolvedScope, scopeId: resolvedScopeId } = targetResolution.target;
            const missingDetails = [];
            const effectiveStatus = status || (issueId && !comment && !externalId ? "Open" : undefined);
            if (comment && !status && !externalId) {
                return {

                isError: true,
            };
        }
    });
    server.registerTool("get-asset-groups", {
        description: getToolDescription("get-asset-groups"),
        inputSchema: {},
    }, async () => {
        try {
            const response = await makeAppScanRequest("/api/v4/AssetGroups");
            const groups = response.Items || response || [];
            if (groups.length === 0) {
                return {
                    content: [
                        {
                            type: "text",
                            text: "No asset groups found",
                        },
                    ],
                };
            }
            const formattedGroups = groups.map((group) => [
                `Name: ${group.Name || "Unknown"}`,
                `ID: ${group.Id || "Unknown"}`,
                `Description: ${group.Description || "No description"}`,
                "---",
            ].join("\n"));
            return {
                content: [
                    {
                        type: "text",
                        text: `Found ${groups.length} asset group(s):\n\n${formattedGroups.join("\n")}`,
                    },
                ],
            };
        }
        catch (error) {
            return {
                content: [
                    {
                        type: "text",
                        text: `Error fetching asset groups: ${error instanceof Error ? error.message : String(error)}`,
                    },
                ],
                isError: true,
            };
        }
    });
    registerCreateScanTool(server);
}
